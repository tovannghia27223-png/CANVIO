function mixHslTriplet(r0,g0,b0,v,out,at){
 let r=r0/255,g=g0/255,b=b0/255,mx=Math.max(r,g,b),mn=Math.min(r,g,b),d=mx-mn,l=(mx+mn)/2;
 if(d<.000001){out[at]=r0;out[at+1]=g0;out[at+2]=b0;return}
 let h=0,ss=d/(1-Math.abs(2*l-1));
 if(mx===r)h=60*(((g-b)/d)%6);else if(mx===g)h=60*((b-r)/d+2);else h=60*((r-g)/d+4);if(h<0)h+=360;
 let dh=0,ds=0,dl=0;
 for(let i=0;i<hslBands.length;i++){
  const band=hslBands[i],vh=+v[band+'Hue']||0,vs=+v[band+'Sat']||0,vl=+v[band+'Lum']||0;if(!vh&&!vs&&!vl)continue;
  const center=hslCenters[band],raw=Math.abs(h-center),dist=Math.min(raw,360-raw);if(dist>=44)continue;
  const weight=(.5+.5*Math.cos(Math.PI*dist/44))*Math.min(1,ss*2.5);
  dh+=vh*.6*weight;ds+=vs*.0055*weight;dl+=vl*.0025*weight;
 }
 if(!dh&&!ds&&!dl){out[at]=r0;out[at+1]=g0;out[at+2]=b0;return}
 h=((h+dh)%360+360)%360;ss=Math.max(0,Math.min(1,ss+ds));l=Math.max(0,Math.min(1,l+dl));
 const c=(1-Math.abs(2*l-1))*ss,hp=h/60,x=c*(1-Math.abs((hp%2)-1));let rr=0,gg=0,bb=0;
 if(hp<1){rr=c;gg=x}else if(hp<2){rr=x;gg=c}else if(hp<3){gg=c;bb=x}else if(hp<4){gg=x;bb=c}else if(hp<5){rr=x;bb=c}else{rr=c;bb=x}
 const mm=l-c/2;out[at]=Math.round(Math.max(0,Math.min(1,rr+mm))*255);out[at+1]=Math.round(Math.max(0,Math.min(1,gg+mm))*255);out[at+2]=Math.round(Math.max(0,Math.min(1,bb+mm))*255);
