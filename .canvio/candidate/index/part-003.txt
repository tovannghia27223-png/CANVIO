    if(!(blob instanceof Blob)||!Number.isFinite(durationMs)||durationMs<=0)return {blob:blob,fixed:false,reason:'invalid-input'};
    function readVint(a,off,isId){
      var first=a[off];if(first===undefined||first===0)return null;var marker=0x80,len=1;
      while(len<=8&&!(first&marker)){marker>>=1;len++}if(len>8||!marker)return null;
      var value=BigInt(isId?first:(first&(marker-1)));
      for(var i=1;i<len;i++){if(a[off+i]===undefined)return null;value=(value<<8n)|BigInt(a[off+i])}
      if(!isId){var unknown=(1n<<BigInt(7*len))-1n;if(value===unknown)value=null}
      return {value:value,length:len};
    }
    function encodeSize(value,preferred){
      value=BigInt(value);var len=Math.max(1,preferred||1);
      function fits(n){return value>=0n&&value<((1n<<BigInt(7*n))-1n)}
      while(len<=8&&!fits(len))len++;if(len>8)throw new Error('WebM segment exceeds EBML size encoding');
      var out=new Uint8Array(len),v=value;for(var i=len-1;i>=0;i--){out[i]=Number(v&255n);v>>=8n}
      out[0]|=1<<(8-len);return out;
    }
    async function bytesAt(offset,length){return new Uint8Array(await blob.slice(offset,offset+length).arrayBuffer())}
    async function elementHeader(offset){var a=await bytesAt(offset,12),id=readVint(a,0,true);if(!id)return null;var size=readVint(a,id.length,false);if(!size)return null;return {id:id.value,size:size.value,idLen:id.length,sizeLen:size.length,headerLen:id.length+size.length};}
    try{
      var top=await elementHeader(0);if(!top||top.id!==0x1A45DFA3n||top.size===null)return {blob:blob,fixed:false,reason:'ebml-header-not-recognized'};
      var segmentOffset=top.headerLen+Number(top.size),seg=await elementHeader(segmentOffset);
      if(!seg||seg.id!==0x18538067n)return {blob:blob,fixed:false,reason:'segment-not-found'};
      var segSizeOffset=segmentOffset+seg.idLen,segPayloadStart=segSizeOffset+seg.sizeLen,pos=segPayloadStart,info=null,scaleNs=1000000n;
      for(var n=0;n<128&&pos<blob.size;n++){
        var child=await elementHeader(pos);if(!child||child.size===null)break;
        var dataStart=pos+child.headerLen,dataEnd=dataStart+Number(child.size);if(dataEnd>blob.size||dataEnd<dataStart)break;
        if(child.id===0x1549A966n){info={offset:pos,dataStart:dataStart,dataEnd:dataEnd,idLen:child.idLen,sizeLen:child.sizeLen,size:child.size};break}
        if(child.id===0x1F43B675n)break;
        pos=dataEnd;
      }
      if(!info)return {blob:blob,fixed:false,reason:'segment-info-not-found-before-cluster'};
      var inner=info.dataStart,hasDuration=false,hasCrc=false;
      while(inner<info.dataEnd){var el=await elementHeader(inner);if(!el||el.size===null)break;var ds=inner+el.headerLen,de=ds+Number(el.size);if(de>info.dataEnd||de<ds)break;
        if(el.id===0x4489n){hasDuration=true;break}if(el.id===0xBFn){hasCrc=true;break}
        if(el.id===0x2AD7B1n&&el.size>0n&&el.size<=8n){var sc=await bytesAt(ds,Number(el.size)),sv=0n;for(var si=0;si<sc.length;si++)sv=(sv<<8n)|BigInt(sc[si]);if(sv>0n)scaleNs=sv;}
        inner=de;
      }
      if(hasDuration)return {blob:blob,fixed:false,reason:'duration-already-present'};
      if(hasCrc)return {blob:blob,fixed:false,reason:'info-crc-present-safe-skip'};
      var durationUnits=durationMs*1000000/Number(scaleNs);if(!Number.isFinite(durationUnits)||durationUnits<=0)return {blob:blob,fixed:false,reason:'duration-unrepresentable'};
      var payload=new Uint8Array(11);payload.set([0x44,0x89,0x88],0);new DataView(payload.buffer).setFloat64(3,durationUnits,false);
      var newInfoSize=BigInt(info.size)+BigInt(payload.length),newInfoSizeBytes=encodeSize(newInfoSize,info.sizeLen),infoSizeOffset=info.offset+info.idLen;
      var infoDelta=(newInfoSizeBytes.length-info.sizeLen)+payload.length;
      var oldSegSize=seg.size===null?BigInt(blob.size-segPayloadStart):seg.size;
      var newSegmentSize=oldSegSize+BigInt(infoDelta),newSegSizeBytes=encodeSize(newSegmentSize,seg.sizeLen);
      var patched=new Blob([
        blob.slice(0,segSizeOffset),newSegSizeBytes,
        blob.slice(segPayloadStart,infoSizeOffset),newInfoSizeBytes,
        blob.slice(info.dataStart,info.dataEnd),payload,
        blob.slice(info.dataEnd)
      ],{type:blob.type||'video/webm'});
      return {blob:patched,fixed:true,reason:'duration-metadata-added',durationMs:durationMs};
    }catch(e){return {blob:blob,fixed:false,reason:'repair-error: '+String(e&&e.message||e)}}
  }

/* 1. Enhance prompt in AI Studio */
async function enhancePrompt(){
  var inp=document.getElementById('studioPrompt');
  if(!inp||!inp.value.trim()){ if(window.CV)CV.toast('Nhập mô tả trước'); return; }
  requireKey();
  var orig=inp.value.trim();
  inp.disabled=true;
  if(window.CV)CV.toast('Đang mở rộng câu lệnh…');
  try{
    var result=await call(
      'Bạn là copywriter sáng tạo cho CANVIO — công cụ thiết kế AI.\n'+
      'Mở rộng câu lệnh sau thành mô tả chi tiết hơn (tiếng Việt, 2-3 câu, nêu ngành, phong cách, cảm giác, màu sắc):\n"'+orig+'"',
      {maxTokens:200}
    );
    inp.value=result.trim()||orig;
    if(window.CV)CV.toast('✦ Đã mở rộng bởi Gemini');
  }catch(e){
    if(e.message==='NO_KEY') return;
    if(window.CV)CV.toast('Lỗi: '+e.message);
  }finally{ inp.disabled=false; }
}

/* 2. AI generate color palette suggestion */
async function suggestPalette(prompt){
  requireKey();
  try{
    var r=await call(
      'Dựa trên mô tả sau, đề xuất bảng màu gồm 2 màu HEX chính (primary, accent) phù hợp nhất. Trả lời JSON: {"primary":"#hex","accent":"#hex","name":"tên bảng màu ngắn"}\n'+
      'Mô tả: '+prompt,
      {maxTokens:80}
    );
    return JSON.parse(r.match(/\{[\s\S]*\}/)[0]);
  }catch(e){ return null; }
}

/* 3. ArtCraft — AI enhance prompt */
async function craftEnhance(){
  var out=document.getElementById('ac-promptOut');
  if(!out||!out.value.trim()){ if(window.CV)CV.toast('Craft prompt trước'); return; }
  requireKey();
  if(window.CV)CV.toast('Gemini đang tinh chỉnh prompt…');
  try{
    var r=await call(
      'Bạn là chuyên gia viết AI image prompt. Tinh chỉnh prompt sau, giữ nguyên ý nghĩa nhưng thêm chi tiết cinematography, lighting và texture. Chỉ trả về prompt đã cải thiện, không giải thích:\n'+out.value,
      {maxTokens:300}
    );
    out.value=r.trim();
    if(window.CV)CV.toast('✦ Prompt đã được tinh chỉnh bởi Gemini');
  }catch(e){
    if(e.message!=='NO_KEY'&&window.CV)CV.toast('Lỗi: '+e.message);
  }
}

/* 4. ShotCraft — AI generate shot list từ script */
async function generateShotList(script){
  requireKey();
  if(window.CV)CV.toast('Gemini đang phân tích kịch bản…');
  try{
    var r=await call(
      'Phân tích kịch bản sau và trả về JSON array các shot. Mỗi shot gồm: type (establishing/medium/closeup/master/pov/cutaway), desc (tiếng Việt, mô tả ngắn hành động), duration (giây, số nguyên 3-10).\n'+
      'Chỉ trả về JSON array, không giải thích.\nKịch bản:\n'+script,
      {maxTokens:600}
    );
    var arr=JSON.parse(r.match(/\[[\s\S]*\]/)[0]);
    return Array.isArray(arr)?arr:[];
  }catch(e){ return []; }
}

function buildPrompt(){
  var mode=el('ac-modeTab')?el('ac-modeTab').querySelector('.on').getAttribute('data-mode'):'compose';
  var parts=[];
  if(mode==='compose'){
    var subj=g('ac-subject')||'a subject';
    var style=g('ac-style');var light=g('ac-light');var angle=g('ac-angle');var ratio=g('ac-ratio');var extra=g('ac-extra');
    parts.push(subj);
    parts.push(style);
    parts.push(light+' lighting');
    parts.push(angle+' shot');
    parts.push('aspect ratio '+ratio);
    if(extra)parts.push(extra);
    parts.push('highly detailed, professional photography, 8K resolution');
  }else if(mode==='scene'){
    var env=g('ac-env');var lens=g('ac-lens');var render=g('ac-render');var tod=g('ac-tod');var weather=g('ac-weather');var props=g('ac-props');
    parts.push(env+' environment');
    parts.push(tod+', '+weather+' weather');
    if(props)parts.push(props.split('\n').join(', '));
    parts.push(lens+' lens, '+render);
    parts.push('cinematic composition, masterful lighting');
  }else{
    return g('ac-wfGoal')||'AI workflow prompt';
  }
  return parts.filter(Boolean).join(', ');
}

async function tileify(it) {
  if (it.svg) return;

    const t = cleanSvg(await (await fetch(it.url)).text()), vb = (t.match(/viewBox="([^"]+)"/) || [0, '0 0 24 24'])[1], inner = t.replace(/^[\s\S]*?<svg[^>]*>/, '').replace(/<\/svg>\s*$/, ''), p = P();

  } catch { /* offline: dùng icon thường */ }
}

const CVZip = (() => {
  const T = new Uint32Array(256);
  for (let n = 0; n < 256; n++) { let c = n; for (let k = 0; k < 8; k++) c = c & 1 ? 0xEDB88320 ^ (c >>> 1) : c >>> 1; T[n] = c >>> 0; }
  const crc32 = u => { let c = 0xFFFFFFFF; for (let i = 0; i < u.length; i++) c = T[(c ^ u[i]) & 255] ^ (c >>> 8); return (c ^ 0xFFFFFFFF) >>> 0; };
  const u8 = async d => d instanceof Blob ? new Uint8Array(await d.arrayBuffer()) : typeof d === 'string' ? new TextEncoder().encode(d) : d;
  async function build(files) {
    const enc = new TextEncoder(), parts = [], cen = []; let off = 0;
    const now = new Date();
    const dt = ((now.getHours() << 11) | (now.getMinutes() << 5) | (now.getSeconds() >> 1)) & 0xFFFF;
    const dd = (((now.getFullYear() - 1980) << 9) | ((now.getMonth() + 1) << 5) | now.getDate()) & 0xFFFF;
    for (const f of files) {
      const data = await u8(f.data), name = enc.encode(f.name), crc = crc32(data);
      const lh = new DataView(new ArrayBuffer(30));
      lh.setUint32(0, 0x04034b50, true); lh.setUint16(4, 20, true); lh.setUint16(6, 0x0800, true); lh.setUint16(8, 0, true);
      lh.setUint16(10, dt, true); lh.setUint16(12, dd, true); lh.setUint32(14, crc, true); lh.setUint32(18, data.length, true);
      lh.setUint32(22, data.length, true); lh.setUint16(26, name.length, true); lh.setUint16(28, 0, true);
      const ch = new DataView(new ArrayBuffer(46));
      ch.setUint32(0, 0x02014b50, true); ch.setUint16(4, 20, true); ch.setUint16(6, 20, true); ch.setUint16(8, 0x0800, true); ch.setUint16(10, 0, true);
      ch.setUint16(12, dt, true); ch.setUint16(14, dd, true); ch.setUint32(16, crc, true); ch.setUint32(20, data.length, true);
      ch.setUint32(24, data.length, true); ch.setUint16(28, name.length, true); ch.setUint32(42, off, true);
      parts.push(new Uint8Array(lh.buffer), name, data);
      cen.push(new Uint8Array(ch.buffer), name);
      off += 30 + name.length + data.length;
    }
    const cdSize = cen.reduce((s, p) => s + p.length, 0);
    const e = new DataView(new ArrayBuffer(22));
    e.setUint32(0, 0x06054b50, true); e.setUint16(8, files.length, true); e.setUint16(10, files.length, true);
    e.setUint32(12, cdSize, true); e.setUint32(16, off, true);
    return new Blob([...parts, ...cen, new Uint8Array(e.buffer)], { type: 'application/zip' });
  }
  return { build, crc32 };
})();

window.CVAudioMix = (() => {
  let ac = null;
  return {
    attach(stream, clips, duration) {
      try {
        const audios = (clips || []).filter(c => c.type === 'audio' && c.el);
        if (!audios.length) return null;
        ac = ac || new (window.AudioContext || window.webkitAudioContext)();
        const dest = ac.createMediaStreamDestination();
        const nodes = audios.map(c => {
          if (!c.el.__cvNode) c.el.__cvNode = ac.createMediaElementSource(c.el);
          if (!c.el.__cvGain) c.el.__cvGain = ac.createGain();
          const baseVolume = Math.max(0, Math.min(1, Number.isFinite(+c.volume) ? +c.volume : 1));
          c.el.__cvGain.gain.cancelScheduledValues(ac.currentTime);
          c.el.__cvGain.gain.setValueAtTime(baseVolume, ac.currentTime);
          c.el.__cvNode.disconnect(); c.el.__cvGain.disconnect();
          const fx=c.audioFx&&c.audioFx.enabled?c.audioFx:null;
          let chain=[];
          if(fx){
            const low=ac.createBiquadFilter();low.type='lowshelf';low.frequency.value=120;low.gain.value=Math.max(-18,Math.min(18,+fx.lowDb||0));
            const mid=ac.createBiquadFilter();mid.type='peaking';mid.frequency.value=Math.max(150,Math.min(8000,+fx.midHz||1000));mid.Q.value=0.8;mid.gain.value=Math.max(-18,Math.min(18,+fx.midDb||0));
            const high=ac.createBiquadFilter();high.type='highshelf';high.frequency.value=8500;high.gain.value=Math.max(-18,Math.min(18,+fx.highDb||0));
            const comp=ac.createDynamicsCompressor();comp.threshold.value=Math.max(-60,Math.min(0,+fx.threshold||-18));comp.ratio.value=Math.max(1,Math.min(20,+fx.ratio||3));comp.attack.value=Math.max(.001,Math.min(1,+fx.attack||.02));comp.release.value=Math.max(.01,Math.min(2,+fx.release||.18));
            const limiter=ac.createDynamicsCompressor();limiter.threshold.value=Math.max(-12,Math.min(0,+fx.limiterDb||-1));limiter.knee.value=0;limiter.ratio.value=20;limiter.attack.value=.003;limiter.release.value=.08;
            chain=[low,mid,high,comp,limiter];
          }
          c.el.__cvNode.connect(c.el.__cvGain);
          if(chain.length){let previous=c.el.__cvGain;chain.forEach(node=>{previous.connect(node);previous=node});previous.connect(dest)}else c.el.__cvGain.connect(dest);
          return {source:c.el.__cvNode,gain:c.el.__cvGain,chain:chain};
        });
        dest.stream.getAudioTracks().forEach(t => stream.addTrack(t));
        const timers = [];
        return {
          start() {
            if (ac.state === 'suspended') ac.resume();
            audios.forEach(c => {
              const s = Math.max(0, +c.start || 0);
              const speed = Math.max(.25, Math.min(4, +c.speed || 1));
              const playable = c.trimOut != null ? Math.min(+c.duration || duration, Math.max(0, +c.trimOut - (+c.trimIn || 0)) / speed) : (+c.duration || duration);
              const base = Math.max(0, Math.min(1, Number.isFinite(+c.volume) ? +c.volume : 1));
              const param = c.el.__cvGain && c.el.__cvGain.gain;
              const envelope = Array.isArray(c.volumeEnvelope) ? c.volumeEnvelope.slice(0, 64).map(p => ({t:Number(p && p.t),v:Number(p && p.v)})).filter(p => Number.isFinite(p.t) && Number.isFinite(p.v) && p.t >= 0 && p.t <= playable).sort((a,b) => a.t-b.t) : [];
              if (param && envelope.length) {
                const startAt = ac.currentTime + s;
                param.cancelScheduledValues(startAt);
                param.setValueAtTime(base, startAt);
                let lastT = -1;
                envelope.forEach((point, index) => {
                  const at = startAt + Math.max(0, Math.min(playable, point.t));
                  const value = base * Math.max(0, Math.min(1, point.v));
                  if (Math.abs(point.t-lastT) < 0.0001) return;
                  if (index === 0 || point.t === 0) param.setValueAtTime(value, at);
                  else param.linearRampToValueAtTime(value, at);
                  lastT = point.t;
                });
              }
              timers.push(setTimeout(() => { try { c.el.currentTime = +c.trimIn || 0; c.el.playbackRate = speed; c.el.play(); } catch {} }, s * 1000));
              timers.push(setTimeout(() => { try { c.el.pause(); } catch {} }, (s + playable) * 1000));
            });
          },
          stop() {
            timers.forEach(clearTimeout);
            audios.forEach(c => { try { c.el.pause(); } catch {} });
            nodes.forEach(n => { try { n.source.disconnect(); n.gain.disconnect(); (n.chain||[]).forEach(node=>{try{node.disconnect()}catch{}}); n.source.connect(n.gain); n.gain.connect(ac.destination); } catch {} });
          }
        };
      } catch (e) { report(e, 'audio-mix'); return null; }
    }
  };
})();

function vet(html) {
  html = String(html).replace(/^[\s\S]*?```(?:html)?\s*\n/i, m => /```/.test(m) ? '' : m).replace(/```[\s\S]*$/, '').trim();
  const i = html.search(/<!doctype html|<html/i); if (i < 0) throw new Error('Không nhận được HTML hợp lệ'); html = html.slice(i);
  if (/\b(fetch|XMLHttpRequest|WebSocket|sendBeacon|eval|importScripts)\s*\(|document\.cookie|window\.top|parent\.|<form[^>]+action\s*=\s*["']?https?/i.test(html)) throw new Error('Mã chứa lệnh mạng/truy cập không được phép');
  html = html.replace(/<meta[^>]+http-equiv\s*=\s*["']?refresh[^>]*>/gi, '').replace(/<script[^>]+src\s*=\s*["']([^"']+)["'][^>]*>\s*<\/script>/gi, (m, u) => /^https:\/\/(cdnjs\.cloudflare\.com|cdn\.jsdelivr\.net|unpkg\.com)\//.test(u) ? m : '');
  return html;
}
