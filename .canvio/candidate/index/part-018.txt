    if (type && type !== 'video/mp4' && type !== 'video/webm') return Promise.reject(new Error('Chỉ kiểm tra được file video MP4 hoặc WebM.'));
    if (!extension && type === 'video/mp4') extension = 'mp4';
    if (!extension && type === 'video/webm') extension = 'webm';
    if (!extension) return Promise.reject(new Error('Không xác định được định dạng MP4/WebM từ tên hoặc MIME.'));
    if (type && ((extension === 'mp4' && type !== 'video/mp4') || (extension === 'webm' && type !== 'video/webm'))) return Promise.reject(new Error('Phần mở rộng và MIME của file không khớp.'));
    if (!window.URL || typeof URL.createObjectURL !== 'function') return Promise.reject(new Error('Object URL không khả dụng để kiểm tra file.'));
    return new Promise(function (resolve, reject) {
      var video = document.createElement('video');
      var url = URL.createObjectURL(file);
      var settled = false;
      var timer = setTimeout(function () { finish(new Error('Hết thời gian chờ trình duyệt đọc metadata/decode file.')); }, Number(options.timeoutMs) > 0 ? Math.min(Number(options.timeoutMs), 30000) : 15000);
      function cleanup() {
        clearTimeout(timer);
        try { video.pause(); } catch (error) {}
        try { video.removeAttribute('src'); video.load(); } catch (error) {}
        try { URL.revokeObjectURL(url); } catch (error) {}
      }
      function finish(error) {
        if (settled) return;
        settled = true;
        if (error) { cleanup(); reject(error); return; }
        var report = {
          format: extension,
          mime: type || 'browser-detected',
          bytes: Number(file.size),
          durationSeconds: Number(video.duration),
          width: Number(video.videoWidth),
          height: Number(video.videoHeight),
          metadataReadable: true,
          decodeReadiness: true,
          scope: { sourceFileOnly: true, upload: false, storageMutation: false, avSyncCertified: false }
        };
        cleanup();
        lastReport = report;
        resolve(JSON.parse(JSON.stringify(report)));
      }
      video.preload = 'auto';
      video.muted = true;
      video.playsInline = true;
      video.onloadedmetadata = function () {
        if (!Number.isFinite(video.duration) || video.duration <= 0 || video.videoWidth <= 0 || video.videoHeight <= 0) {
          finish(new Error('Metadata video không hợp lệ: thiếu thời lượng hoặc kích thước khung hình.'));
          return;
        }
        // `canplay` is a local browser decode-readiness signal, not a complete
        // frame-by-frame or audio/video synchronization certification.
      };
      video.oncanplay = function () { finish(null); };
      video.onerror = function () { finish(new Error('Trình duyệt không giải mã được file đã chọn.')); };
      try { video.src = url; video.load(); } catch (error) { finish(error); }
    });
  }
  function capabilities() {
    var api = window.CV && CV.engines && CV.engines.video;
    var value = api && typeof api.exportCapabilities === 'function' ? api.exportCapabilities() : null;
    return { browser: value || null, mp4IsOptional: true, requiredRealMediaAcceptance: true };
  }
  window.CANVIOExportAcceptance = {
    version: '58.0.0',
    inspectFile: inspectFile,

/* CANVIO P60 — In-app accessibility screening; it is not a WCAG certification. */
(function () {
  'use strict';
  var lastReport = null;
  function visible(node) {
    var current = node;
    while (current && current !== document) {
      if (current.hidden || (current.getAttribute && current.getAttribute('aria-hidden') === 'true')) return false;
      var style = window.getComputedStyle ? window.getComputedStyle(current) : null;
      if (style && (style.display === 'none' || style.visibility === 'hidden')) return false;
      current = current.parentElement;
    }
    return !!node;
  }
  function hasName(node) {
    var aria = (node.getAttribute('aria-label') || '').trim();
    var labelled = (node.getAttribute('aria-labelledby') || '').trim();
    var title = (node.getAttribute('title') || '').trim();
    var text = (node.innerText || node.textContent || '').replace(/\s+/g, ' ').trim();
    if (aria || title || text) return true;
    if (labelled && document.getElementById(labelled)) return true;
    if (node.id && document.querySelector && document.querySelector('label[for="' + String(node.id).replace(/"/g, '\\"') + '"]')) return true;
    return !!(node.closest && node.closest('label'));
  }
  function inspect(root) {
    root = root || document;
    var issues = [], totalIssues = 0, counts = { controls: 0, unnamedControls: 0, images: 0, imagesMissingAlt: 0, landmarks: 0, headings: 0, dialogs: 0, dialogsMissingName: 0, buttonsMissingType: 0 };
    function add(code, node, detail) {
      totalIssues++;
      if (issues.length < 200) issues.push({ code: code, tag: node && node.tagName ? node.tagName.toLowerCase() : '', id: node && node.id || '', detail: detail });
    }
    var controls = root.querySelectorAll ? root.querySelectorAll('button,a[href],input,select,textarea,[role="button"],[role="link"],[tabindex]:not([tabindex="-1"])') : [];
    Array.prototype.forEach.call(controls, function (node) {
      if (!visible(node)) return;
      counts.controls++;
      if (!hasName(node)) { counts.unnamedControls++; add('control-name-missing', node, 'Thiếu tên truy cập bằng text, label hoặc ARIA.'); }
      if (node.tagName && node.tagName.toLowerCase() === 'button' && node.form && !node.hasAttribute('type')) { counts.buttonsMissingType++; add('button-type-missing', node, 'Nên khai báo type="button" hoặc type="submit".'); }
    });
    var images = root.querySelectorAll ? root.querySelectorAll('img') : [];
    Array.prototype.forEach.call(images, function (node) {
      counts.images++;
      if (!node.hasAttribute('alt')) { counts.imagesMissingAlt++; add('image-alt-missing', node, 'Ảnh cần alt mô tả hoặc alt="" nếu trang trí.'); }
    });
    var landmarks = root.querySelectorAll ? root.querySelectorAll('main,nav,header,footer,aside,[role="main"],[role="navigation"],[role="region"]') : [];
    counts.landmarks = landmarks.length;
    var headings = root.querySelectorAll ? root.querySelectorAll('h1,h2,h3,h4,h5,h6') : [];
    counts.headings = headings.length;
    var dialogs = root.querySelectorAll ? root.querySelectorAll('[role="dialog"],[aria-modal="true"]') : [];
    counts.dialogs = dialogs.length;
    Array.prototype.forEach.call(dialogs, function (node) {
      if (!hasName(node)) { counts.dialogsMissingName++; add('dialog-name-missing', node, 'Dialog cần accessible name.'); }
    });
    var report = {
      version: '60.0.0',
      counts: counts,
      issueCount: totalIssues,
      capturedIssues: issues.length,
      issues: issues,
      truncated: totalIssues > issues.length,
      scope: { automatedScreeningOnly: true, wcagCertified: false, true200PercentZoomTested: false, screenReaderTested: false }
    };
    lastReport = report;
    return JSON.parse(JSON.stringify(report));
  }
  window.CANVIOAccessibilityAudit = { version: '60.0.0', inspect: inspect, last: function () { return lastReport ? JSON.parse(JSON.stringify(lastReport)) : null; } };
  var suite = window.CANVIOPhaseSuite;
  if (suite) {

      var result = previousTests ? previousTests.apply(this, arguments) : {};

      return result;
    };

  function installPalette() {
    var cv = window.CV;
    if (!cv || !cv.palette || typeof cv.palette.build !== 'function' || cv.palette.build.__p90p96) return false;
    var originalBuild = cv.palette.build;
    cv.palette.build = function () {
      var items = originalBuild.apply(this, arguments) || [];
      var extra = commandItems();
      extra.forEach(function (item) {
        if (!items.some(function (existing) { return existing && existing.label === item.label; })) items.unshift(item);
      });
      return items;
    };
    cv.palette.build.__p90p96 = true;

    var originalRender = cv.palette.render;
    if (typeof originalRender === 'function' && !originalRender.__p90p96) {
      var enhancedRender = function (query) {
        var result = originalRender.apply(this, arguments);
        var list = $('#paletteList');
        if (!list) return result;
        list.setAttribute('role', 'listbox');
        list.setAttribute('aria-label', 'Kết quả lệnh CANVIO');
        $$('.palItem', list).forEach(function (node) {
          var index = Number(node.getAttribute('data-i'));
          var item = cv.palette.items && cv.palette.items[index];
          node.setAttribute('role', 'option');
          node.setAttribute('tabindex', '-1');
          node.setAttribute('aria-selected', node.classList.contains('on') ? 'true' : 'false');
          if (item && item.shortcut && !node.querySelector('.cvux-pal-shortcut')) {
            var chip = document.createElement('kbd');
            chip.className = 'cvux-pal-shortcut';
            chip.textContent = item.shortcut;
            node.appendChild(chip);
          }
        });
        return result;
      };
      enhancedRender.__p90p96 = true;
      cv.palette.render = enhancedRender;
    }
    var input = $('#paletteInput');
    if (input && !input.__p90p96) {
      input.__p90p96 = true;
      input.setAttribute('aria-label', 'Tìm lệnh CANVIO bằng tiếng Việt hoặc tiếng Anh');
      input.setAttribute('autocomplete', 'off');
      input.addEventListener('input', function () { try { cv.palette.render(input.value); } catch (e) {} });
      input.addEventListener('keydown', function (event) {
        var list = $('#paletteList');
        var nodes = list ? $$('.palItem', list) : [];
        if (event.key === 'Escape') { event.preventDefault(); cv.palette.close(); return; }
        if (!nodes.length) return;
        if (event.key === 'ArrowDown' || event.key === 'ArrowUp') {
          event.preventDefault();
          var active = nodes.findIndex(function (n) { return n.classList.contains('on'); });
          var next = event.key === 'ArrowDown' ? (active + 1 + nodes.length) % nodes.length : (active - 1 + nodes.length) % nodes.length;
          nodes.forEach(function (node, i) { node.classList.toggle('on', i === next); node.setAttribute('aria-selected', i === next ? 'true' : 'false'); });
          nodes[next].scrollIntoView({ block: 'nearest' });
        } else if (event.key === 'Enter') {
          var activeNode = nodes.find(function (n) { return n.classList.contains('on'); }) || nodes[0];
          if (activeNode) { event.preventDefault(); event.stopImmediatePropagation(); activeNode.click(); }
        }
      }, true);
    }
    var palette = $('#palette');
    if (palette) { palette.setAttribute('role', 'dialog'); palette.setAttribute('aria-modal', 'true'); palette.setAttribute('aria-label', 'Bảng lệnh CANVIO'); }
    var originalOpen = cv.palette.open;
    if (typeof originalOpen === 'function' && !originalOpen.__p90p96) {
      var enhancedOpen = function () {
        var result = originalOpen.apply(this, arguments);
        if (input) { input.value = ''; try { cv.palette.render(''); } catch (e) {} }
        return result;
      };
      enhancedOpen.__p90p96 = true;
      cv.palette.open = enhancedOpen;
    }
    return true;
  }

  function planWordCues() {
    var v = api(), s = state(), sel = $('#cv99-cue-select'), clip = selectedCaption(sel && sel.value);
    if (!v || !s || !clip) return { ok: false, reason: 'Chọn một cue phụ đề trên track chữ trước.' };
    if (isLocked(v, clip) || (v.trackState && v.trackState(3).locked)) return { ok: false, reason: 'Track chữ hoặc cue đang khóa.' };
    var words = String(clip.text || '').trim().split(/\s+/).filter(Boolean);
    if (words.length < 2) return { ok: false, reason: 'Cue cần có ít nhất hai từ.' };
    var duration = Number(clip.duration), weights = words.map(function (w) { return Math.max(1, w.replace(/[^\p{L}\p{N}]/gu, '').length) + 0.35; });
    var total = weights.reduce(function (a, b) { return a + b; }, 0), cursor = Number(clip.start), generated = [];
    words.forEach(function (word, i) {
      var d = i === words.length - 1 ? Number(clip.start) + duration - cursor : duration * weights[i] / total;
      generated.push({ text: word, start: cursor, duration: d }); cursor += d;
    });
    if (generated.some(function (w) { return !Number.isFinite(w.duration) || w.duration < 0.2; })) return { ok: false, reason: 'Cue quá ngắn để tạo từng từ hợp lệ (Video Engine yêu cầu clip chữ tối thiểu 0,2 giây). Hãy dùng cue dài hơn hoặc giữ phụ đề theo câu.' };
    return { ok: true, clipId: clip.id, name: clip.name, cue: String(clip.text), words: generated, before: snapshot(), note: 'Ước lượng theo độ dài từ trong timestamp của cue; không phải căn chỉnh âm học từng từ.' };
  }

  function normalizeExportOptions(input) {
    input = input || {};
    var presets = { '480': [854, 480], '720': [1280, 720], '1080': [1920, 1080], '1440': [2560, 1440], '2160': [3840, 2160] };
    var p = presets[String(input.resolution || '1080')] || presets['1080'];
    var fps = [24, 25, 30, 50, 60].indexOf(Number(input.fps)) >= 0 ? Number(input.fps) : 30;
    var mbps = Math.max(1, Math.min(20, Number(input.bitrateMbps) || 6));
    return { width: p[0], height: p[1], fps: fps, videoBitsPerSecond: Math.round(mbps * 1000000), resolution: String(input.resolution || '1080') };
  }
  function inspectExportBlob(blob, expected, expectedDuration) {
    return new Promise(function (resolve) {
      if (!(blob instanceof Blob) || !blob.size) { resolve({ ok: false, reason: 'Không có dữ liệu video để kiểm tra.', bytes: blob && blob.size || 0 }); return; }
      var url = URL.createObjectURL(blob), video = document.createElement('video'), timer, settled = false;
      video.muted = true; video.preload = 'metadata';
      function finish(reason) {
        if (settled) return; settled = true; clearTimeout(timer);
        var duration = Number(video.duration), mimeOK = expected === 'mp4' ? /^video\/mp4(?:;|$)/i.test(blob.type) : /^video\/webm(?:;|$)/i.test(blob.type);
        var durationOK = Number.isFinite(duration) && duration > 0 && (!Number.isFinite(expectedDuration) || Math.abs(duration - expectedDuration) <= Math.max(1, expectedDuration * 0.08));
        resolve({ ok: mimeOK && durationOK && reason === 'canplay', mime: blob.type || '', mimeOK: mimeOK, bytes: blob.size, duration: Number.isFinite(duration) ? duration : null, durationOK: durationOK, canplay: reason === 'canplay', level: 'LOCAL_METADATA_AND_CANPLAY_SCREEN_NOT_FULL_FILE_OR_AV_SYNC_CERTIFICATION', note: reason === 'canplay' ? 'Browser nhận metadata và giải mã được tới canplay; chưa chứng nhận toàn file hoặc A/V sync.' : 'Không đạt kiểm tra canplay trong giới hạn thời gian.' });
        try { video.pause(); video.removeAttribute('src'); video.load(); } catch (e) {}
        URL.revokeObjectURL(url);
      }
      video.addEventListener('loadedmetadata', function () { if (video.readyState >= 3) finish('canplay'); }, { once: true });
      video.addEventListener('canplay', function () { finish('canplay'); }, { once: true });
      video.addEventListener('error', function () { finish('decode-error'); }, { once: true });
      timer = setTimeout(function () { finish('timeout'); }, 8000);
      video.src = url; try { video.load(); } catch (e) { finish('load-error'); }
    });
  }
  function setExportStatus(message) { var n = $('#cv101-status'); if (n) n.textContent = String(message || ''); }
  async function runDetailedExport() {
    var v = api(); if (!v || typeof v.exportWebM !== 'function' || typeof v.exportMP4 !== 'function') { setExportStatus('Video Engine chưa cung cấp đường xuất hiện có.'); return false; }
    if (exportBusy) return false;
    var format = $('#cv101-format').value, cap = v.exportCapabilities ? v.exportCapabilities() : {}, webDecision = v.webCodecsMP4Decision ? v.webCodecsMP4Decision() : { ok: false };
    var canMP4 = !!cap.h264Supported || (!!cap.webCodecsAvailable && !!webDecision.ok);
    if (format === 'mp4' && !canMP4) { setExportStatus(webDecision.reason || 'Trình duyệt chưa xác nhận MP4/H.264. Hãy chọn WebM; CANVIO không giả đuôi MP4.'); return false; }
    var opt = normalizeExportOptions({ resolution: $('#cv101-resolution').value, fps: $('#cv101-fps').value, bitrateMbps: $('#cv101-bitrate').value });
    var deviceMemory = Number(navigator.deviceMemory) || 0;
    if (opt.resolution === '2160' && deviceMemory && deviceMemory <= 4 && !window.confirm('Thiết bị báo khoảng ' + deviceMemory + ' GB RAM. Xuất 4K có thể gây tràn bộ nhớ hoặc treo tab. Bạn vẫn muốn tiếp tục?')) return false;
    exportBusy = true; $('#cv101-export').disabled = true;
    setExportStatus('Đang xuất ' + format.toUpperCase() + ' · ' + opt.width + '×' + opt.height + ' · ' + opt.fps + ' fps…');
    try {
      var exportOptions = { width: opt.width, height: opt.height, fps: opt.fps, videoBitsPerSecond: opt.videoBitsPerSecond };
      var blob = await (format === 'mp4' ? v.exportMP4(exportOptions) : v.exportWebM(exportOptions));
      if (!blob) { setExportStatus('Không tạo được file. Kiểm tra trạng thái render hoặc thử WebM/độ phân giải thấp hơn.'); return false; }
      var qc = await inspectExportBlob(blob, format, Number((state() && state().duration) || 0));
      if ($('#cv101-srt').checked && typeof v.srtExport === 'function') v.srtExport();
      if ($('#cv101-vtt').checked && typeof v.vttExport === 'function') v.vttExport();
      setExportStatus((qc.ok ? 'QC cục bộ đạt' : 'QC cục bộ chưa đạt') + ' · ' + (blob.size / 1048576).toFixed(2) + ' MB · ' + (qc.duration ? qc.duration.toFixed(2) + 's' : 'không đọc được duration') + '. ' + qc.note);
      return qc;
    } catch (e) { setExportStatus('Xuất thất bại: ' + (e && e.message || e) + '. Hãy thử lại hoặc chọn WebM/độ phân giải thấp hơn.'); return false; }
    finally { exportBusy = false; $('#cv101-export').disabled = false; }
  }
