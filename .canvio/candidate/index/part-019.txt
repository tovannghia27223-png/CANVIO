  }
  function pathRect(x, y, w, h, rx, H, scale, rot) {

  }

  function pdfFunction(stops, objects) {

        for (var c = 0; c < centers.length; c++) { var q = centers[c], d = (p[0]-q[0])**2 + (p[1]-q[1])**2 + (p[2]-q[2])**2; if (d < min) { min = d; ix = c; } }
        assign[s] = ix; counts[ix]++; sums[ix][0] += p[0]; sums[ix][1] += p[1]; sums[ix][2] += p[2];
      }
      centers = centers.map(function (old, ix) { return counts[ix] ? sums[ix].map(function (v) { return Math.round(v / counts[ix]); }) : old; });
    }
    var rows = centers.map(function (c, i) { return { rgb: c, count: counts[i], share: counts[i] / samples.length, hex: '#' + c.map(function (v) { return clamp(Math.round(v), 0, 255).toString(16).padStart(2, '0'); }).join('').toUpperCase() }; }).filter(function (x) { return x.count > 0; });
    rows.sort(function (a, b) { return b.count - a.count; });
    return { palette: rows, samples: samples.length };
  }
  function luminance(rgb) { var v = rgb.map(function (x) { x = x / 255; return x <= .04045 ? x / 12.92 : Math.pow((x + .055) / 1.055, 2.4); }); return .2126*v[0] + .7152*v[1] + .0722*v[2]; }
  function chroma(rgb) { return Math.max.apply(null, rgb) - Math.min.apply(null, rgb); }
  function suggestPalette(rows) {
    if (!rows || !rows.length) throw new Error('Chưa có bảng màu để gợi ý.');
    var sorted = rows.slice().sort(function (a, b) { return b.count - a.count; });
    var light = sorted.filter(function (x) { return luminance(x.rgb) >= .55; });
    var dark = sorted.filter(function (x) { return luminance(x.rgb) < .25; });
    var dominant = sorted[0];
    var bg = (luminance(dominant.rgb) > .45 ? (light[0] || dominant) : (dark[0] || dominant));
    var ink = sorted.slice().sort(function (a, b) { return Math.abs(luminance(b.rgb) - luminance(bg.rgb)) - Math.abs(luminance(a.rgb) - luminance(bg.rgb)); })[0] || sorted[0];
    var vivid = sorted.filter(function (x) { return x !== bg && x !== ink; }).sort(function (a, b) { return chroma(b.rgb) * Math.sqrt(b.share) - chroma(a.rgb) * Math.sqrt(a.share); });
    var pick = function (i, fb) { return vivid[i] && vivid[i].hex || fb; };
    return { primary: pick(0, dominant.hex), secondary: pick(1, pick(0, dominant.hex)), accent: pick(2, pick(1, dominant.hex)), bg: bg.hex, ink: ink.hex, muted: pick(3, ink.hex), backgroundLuminance: luminance(bg.rgb) };
  }
  async function analyseImage(file) {
    if (!file) throw new Error('Chưa chọn ảnh.');
    if (!/^image\/(png|jpeg|webp)$/i.test(file.type || '')) throw new Error('Chỉ hỗ trợ PNG, JPEG hoặc WebP.');
    if (file.size > 20 * 1024 * 1024) throw new Error('Ảnh vượt giới hạn 20 MiB.');
    if (typeof createImageBitmap !== 'function') throw new Error('Trình duyệt không hỗ trợ giải mã ảnh cục bộ.');
    var bitmap = await createImageBitmap(file);
    try {
      if (bitmap.width < 16 || bitmap.height < 16 || bitmap.width > 6000 || bitmap.height > 6000 || bitmap.width * bitmap.height > 36000000) throw new Error('Kích thước ảnh phải từ 16 px đến 6000 px mỗi chiều và tối đa 36 megapixel.');
      var scale = Math.min(1, 144 / Math.max(bitmap.width, bitmap.height));
      var w = Math.max(1, Math.round(bitmap.width * scale)), h = Math.max(1, Math.round(bitmap.height * scale));
      var canvas = document.createElement('canvas'); canvas.width = w; canvas.height = h;
      var ctx = canvas.getContext('2d', { willReadFrequently: true }); if (!ctx) throw new Error('Không khởi tạo được bộ đọc pixel.');
      ctx.drawImage(bitmap, 0, 0, w, h); var image = ctx.getImageData(0, 0, w, h), q = quantizePixels(image.data, 6);
      var dna = suggestPalette(q.palette), bg = hexRgb(dna.bg), bbox = { x0: w, y0: h, x1: -1, y1: -1 }, changed = 0;
      for (var y = 0; y < h; y++) for (var x = 0; x < w; x++) {
        var ix = (y * w + x) * 4, r = image.data[ix], g = image.data[ix + 1], b = image.data[ix + 2];
        var dist = Math.sqrt((r-bg[0])**2 + (g-bg[1])**2 + (b-bg[2])**2);
        if (dist > 36) { changed++; bbox.x0 = Math.min(bbox.x0, x); bbox.y0 = Math.min(bbox.y0, y); bbox.x1 = Math.max(bbox.x1, x); bbox.y1 = Math.max(bbox.y1, y); }
      }
      var box = bbox.x1 >= bbox.x0 ? { x: bbox.x0 / w, y: bbox.y0 / h, width: (bbox.x1-bbox.x0+1) / w, height: (bbox.y1-bbox.y0+1) / h } : null;
      return { fileName: String(file.name || 'image').slice(0, 120), width: bitmap.width, height: bitmap.height, sampledWidth: w, sampledHeight: h, samples: q.samples, palette: q.palette, suggested: dna, contentBoundsEstimate: box, contentCoverageEstimate: changed / (w*h), limitation: 'Phân tích màu và mật độ pixel ở mức ước lượng. Không OCR, không xác định font chính xác, không suy ra khoảng cách thiết kế chính xác và không tái tạo ảnh y hệt.' };
    } finally { bitmap.close && bitmap.close(); }
  }

  function validateAvcC(avcC) {
    if (!(avcC instanceof Uint8Array)) avcC = new Uint8Array(avcC || 0);
    if (avcC.byteLength < 7 || avcC[0] !== 1) throw new Error('WebCodecs không trả AVCDecoderConfigurationRecord hợp lệ.');
    var lengthSize = (avcC[4] & 3) + 1;
    if (lengthSize !== 4) throw new Error('AVC NAL length không tương thích với muxer MP4 P79.');
    var spsCount = avcC[5] & 31;
    if (!spsCount) throw new Error('AVC configuration thiếu SPS.');
    var off = 6;
    for (var i = 0; i < spsCount; i++) { if (off + 2 > avcC.length) throw new Error('AVC SPS bị cắt.'); var n = (avcC[off] << 8) | avcC[off + 1]; off += 2 + n; if (off > avcC.length) throw new Error('AVC SPS vượt giới hạn.'); }
    if (off >= avcC.length) throw new Error('AVC configuration thiếu PPS count.');
    var ppsCount = avcC[off++]; if (!ppsCount) throw new Error('AVC configuration thiếu PPS.');
    for (var j = 0; j < ppsCount; j++) { if (off + 2 > avcC.length) throw new Error('AVC PPS bị cắt.'); var pn = (avcC[off] << 8) | avcC[off + 1]; off += 2 + pn; if (off > avcC.length) throw new Error('AVC PPS vượt giới hạn.'); }
    return avcC;
  }
  function sampleHasNal(sample, wanted) {

    return false;
  }

    });

'Khoảng 1.2–3 giây, tùy độ dài câu' : 'Khoảng 2–5 giây, tùy độ dài câu', easing: 'ease-out', easingLabel: 'Ease Out cho tiêu đề; phụ đề thường nên đứng yên', order: 'Giữ subtitle trên track 3; tiêu đề/đồ họa bổ sung ở overlay track 4.', guidance: 'Không animate từng từ nếu chưa có đồng bộ lời nói đã xác minh. Ưu tiên tương phản, vùng an toàn và thời gian đọc.' };
    } else if (type === 'audio') {
      item = { label: 'Audio / voice / music', duration: duration, durationLabel: 'Giữ theo nội dung audio', easing: 'none', easingLabel: 'Không dùng keyframe transform', order: 'Voice và nhạc ở track 2; kiểm tra mute/lock và envelope hiện có.', guidance: 'Dùng fade/volume envelope hiện có; tránh cắt âm đột ngột. Không thay đổi gain hoặc điểm cắt tự động.' };
    } else {
      item = { label: 'Clip chưa phân loại', duration: duration, durationLabel: 'Giữ thời lượng hiện tại', easing: 'linear', easingLabel: 'Chưa đủ ngữ cảnh', order: 'Kiểm tra loại media và track trước khi thêm chuyển động.', guidance: 'Không có gợi ý tự động cho clip chưa phân loại.' };
    }
    item.currentDuration = duration;
    item.currentEasing = easing;
    item.currentKeyframes = keyframes;
    item.type = type;
    item.isLocked = !!clip.locked;
    return item;
  }

  function recommendWebsite(component) {
    var tag = '', attrs = {}, classes = '';
    try {
      if (component && typeof component.get === 'function') tag = clean(component.get('tagName') || component.get('type') || '', 40).toLowerCase();
      if (component && typeof component.getAttributes === 'function') attrs = component.getAttributes() || {};
      classes = clean(attrs.class || '', 180).toLowerCase();
    } catch (e) { /* keep safe generic recommendation */ }
    var kind = 'section', result;
    if (/^(h1|h2|h3|h4|h5|h6)$/.test(tag) || /title|heading/.test(classes)) kind = 'heading';
    else if (tag === 'button' || /btn|button|cta/.test(classes)) kind = 'button';
    else if (tag === 'img' || /image|hero-media/.test(classes)) kind = 'image';
    else if (tag === 'p' || /text|paragraph|description/.test(classes)) kind = 'text';
    if (kind === 'heading') result = { kind: kind, label: 'Tiêu đề', duration: 650, easing: 'ease-out', delay: 0, advice: 'Vào nhanh, dừng gọn; tránh overshoot làm giảm độ ổn định thị giác.' };
    else if (kind === 'button') result = { kind: kind, label: 'Nút hành động', duration: 220, easing: 'ease-out', delay: 0, advice: 'Phản hồi ngắn, rõ; tránh delay để không làm chậm thao tác.' };
    else if (kind === 'image') result = { kind: kind, label: 'Ảnh / hero', duration: 850, easing: 'ease-in-out', delay: 0, advice: 'Chuyển động chậm và nhẹ; không giấu nội dung khi reduced-motion hoặc JavaScript không chạy.' };
    else if (kind === 'text') result = { kind: kind, label: 'Đoạn văn', duration: 550, easing: 'ease-out', delay: 0, advice: 'Giữ nhịp đọc; chỉ dùng fade/slide nhẹ, tránh animate từng dòng.' };
    else result = { kind: kind, label: 'Section / container', duration: 700, easing: 'ease-in-out', delay: 0, advice: 'Dùng một hiệu ứng nhất quán cho nhóm; tránh xếp nhiều hiệu ứng cạnh tranh.' };
    result.tag = tag || 'component';
    return result;
  }

  function videoMarkup(clip) {
    var r = recommendVideo(clip);
    var video = window.CV && CV.engines && CV.engines.video;
    var locked = !!(r.isLocked || (video && typeof video.isClipLocked === 'function' && video.isClipLocked(clip)));
    return '<section id="cv78-motion-advice" class="cv78-advice" data-locked="' + (locked ? 'true' : 'false') + '" aria-label="Checklist chuyển động gợi ý">' +
      '<div class="cv78-head"><div><b>Gợi ý chuyển động · P78</b><small>Tham khảo theo loại clip · không tự sửa timeline</small></div><span class="cv78-chip">' + esc(r.label) + '</span></div>' +
      '<div class="cv78-grid"><div><small>Thời lượng tham khảo</small><b>' + esc(r.durationLabel) + '</b></div><div><small>Easing gợi ý</small><b>' + esc(r.easingLabel) + '</b></div><div><small>Keyframe hiện tại</small><b>' + String(r.currentKeyframes) + ' · ' + esc(r.currentEasing) + '</b></div></div>' +

      '<div class="cv78-foot"><span>' + (locked ? 'Track/clip đang khóa: mở khóa mới chỉnh được.' : 'Gợi ý chỉ điền vào biểu mẫu; cần bấm Áp dụng để ghi vào clip.') + '</span>' +
      '<button type="button" class="btn btn-soft sm" data-cv78-fill-video' + (locked ? ' disabled' : '') + '>Điền gợi ý vào biểu mẫu</button></div>' +
      '<div class="cv78-feedback" id="cv78-video-feedback" role="status" aria-live="polite"></div></section>';
  }

  function mountVideoAdvice() {
    var box = $('#vtProps');
    if (!box) return false;
    var state = window.CV && CV.engines && CV.engines.video && typeof CV.engines.video._state === 'function' ? CV.engines.video._state() : null;
    var clip = state && Array.isArray(state.clips) ? state.clips.find(function (x) { return x && x.id === state.selected; }) : null;
    var existing = $('#cv78-motion-advice', box);
    if (!clip) { if (existing) existing.remove(); return false; }
    var locked = !!(clip.locked || (CV.engines.video.isClipLocked && CV.engines.video.isClipLocked(clip)));
    if (existing && existing.dataset.locked === String(locked)) return true;
    var holder = document.createElement('div'); holder.innerHTML = videoMarkup(clip);
    var node = holder.firstElementChild;
    if (existing) existing.replaceWith(node); else if (node) box.appendChild(node);
    return !!node;
  }
