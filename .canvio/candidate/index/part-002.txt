var Templates = (function(){
  var built=null;
  function build(){
    if(built) return built;
    built = TEMPLATE_DEFS.map(function(d,i){
      var size = RATIOS[d.r]||RATIOS.portrait;
      var sample = POSTER_SAMPLE_DATA[d.n];
      var doc = sample ? buildSamplePoster(d,size,sample) : Designer.build({
        w:size.w, h:size.h, ratio:d.r, palette:d.pal, style:d.st, industry:d.ind, variant:d.v,
        name:d.n, brandName: 'NORTHSTAR'
      });
      return {id:'tpl'+i, name:d.n, industry:d.ind, style:d.st, ratio:d.r, palette:d.pal, sample:sample||null, doc:doc};
    });
    return built;
  }
  function render(){
    var grid=document.getElementById('tplGrid');
    if(!grid) return;
    var ind=document.getElementById('tplIndustry');
    var sty=document.getElementById('tplStyle');
    var rat=document.getElementById('tplRatio');
    var fi=ind?ind.value:'all', fs=sty?sty.value:'all', fr=rat?rat.value:'all';
    var list=build().filter(function(t){
      return (fi==='all'||t.industry===fi) && (fs==='all'||t.style===fs) && (fr==='all'||t.ratio===fr);
    });
    var cnt=document.getElementById('tplCount');
    if(cnt) cnt.textContent=list.length+' mẫu';
    if(!list.length){ grid.innerHTML='<div class="emptyState" style="grid-column:1/-1"><div class="ei">▤</div><b>Không có mẫu phù hợp</b><div class="tiny">Thử bỏ bớt bộ lọc.</div></div>'; return; }
    grid.innerHTML = list.map(function(t){
      return '<div class="pvCard tplCard" data-tpl="'+t.id+'">'
        + '<div class="pvThumb" style="height:184px">'+Preview.thumb(t.doc, 240, 184)+'</div>'
        + '<div class="pvBody"><div><b>'+U.esc(t.name)+'</b><small>'+U.esc(indLabel(t.industry))+' · '+U.esc(styleLabel(t.style))+'</small></div>'
        + '<button class="btn btn-soft sm" onclick="Templates.use(\''+t.id+'\')">Dùng</button></div></div>';
    }).join('');
    var cards=grid.querySelectorAll('.tplCard');
    for(var i=0;i<cards.length;i++){
      cards[i].addEventListener('click', function(e){
        if(e.target.closest('button,select,[data-cv190-poster-action]')) return;
        Templates.use(this.getAttribute('data-tpl'));
      });
    }
  }
  function indLabel(k){ return (Designer.INDUSTRY[k]||{}).label || k; }
  function styleLabel(k){ for(var i=0;i<STYLE_LIB.length;i++) if(STYLE_LIB[i].id===k) return STYLE_LIB[i].name; return k; }
  function use(id){
    var t=null, all=build();
    for(var i=0;i<all.length;i++) if(all[i].id===id) t=all[i];
    if(!t) return;
    var clone = JSON.parse(JSON.stringify(t.doc));
    clone.id = U.uid('doc');
    clone.layers.forEach(function(L){ L.id=U.uid('ly'); });
    Poster.setDoc(clone);
    go('poster');
    toast('Đã mở mẫu “'+t.name+'”');
  }
  function random(){
    var all=build();
    var t=all[Math.floor(Math.random()*all.length)];
    use(t.id);
  }
  function fillFilters(){
    var ind=document.getElementById('tplIndustry');
    if(ind && ind.options.length<2){
      Object.keys(Designer.INDUSTRY).forEach(function(k){
        var o=document.createElement('option'); o.value=k; o.textContent=Designer.INDUSTRY[k].label; ind.appendChild(o);
      });
    }
    var sty=document.getElementById('tplStyle');
    if(sty && sty.options.length<2){
      STYLE_LIB.forEach(function(s){
        var o=document.createElement('option'); o.value=s.id; o.textContent=s.name; sty.appendChild(o);
      });
    }
  }
  return {render:render, use:use, random:random, build:build, fillFilters:fillFilters};
})();

/* ============================================================

   ============================================================ */
var Project = (function(){
  function autosave(){
    var doc = Poster.doc;
    if(!doc) return;
    LS.set('currentDoc', doc);
    LS.set('currentName', doc.name||'Thiết kế');
    markSaved();
  }
  var markSaved = U.debounce(function(){
    var el=document.getElementById('saveState');
    if(!el) return;
    el.innerHTML='<i></i><span>Đã lưu</span>';
  }, 300);

  function save(notify){
    var doc=Poster.doc;
    if(!doc){ if(notify) toast('Chưa có gì để lưu'); return; }
    LS.set('currentDoc', doc);
    var list = LS.get('projects', []);
    var found=false;
    list.forEach(function(p){ if(p.id===doc.id){ p.savedAt=new Date().toISOString(); p.doc=doc; found=true; } });
    if(!found) list.unshift({id:doc.id, name:doc.name, savedAt:new Date().toISOString(), doc:doc});
    list = list.slice(0,24);
    LS.set('projects', list);
    Board.refresh();
    if(notify) toast('Đã lưu dự án');
  }

  function createVersion(){
    var doc=Poster.doc;
    if(!doc){ toast('Chưa có thiết kế'); return; }
    var versions = LS.get('versions', []);
    versions.unshift({id:U.uid('v'), at:new Date().toISOString(), name:doc.name, doc:JSON.parse(JSON.stringify(doc))});
    LS.set('versions', versions.slice(0,30));
    renderVersions();
    toast('Đã lưu phiên bản');
  }

  function renderVersions(){
    var box=document.getElementById('versionList');
    if(!box) return;
    var versions=LS.get('versions', []);
    if(!versions.length){ box.innerHTML='<div class="emptyState"><div class="ei">▤</div><b>Chưa có phiên bản</b><div class="tiny">Lưu phiên bản để có thể quay lại bất cứ lúc nào.</div></div>'; return; }
    box.innerHTML = versions.map(function(v){
      return '<div class="listRow"><span class="avatarSq" style="background:var(--brand-soft);color:var(--brand)">▤</span>'
        + '<span style="flex:1"><b>'+U.esc(v.name||'Thiết kế')+'</b><br><small class="tiny">'+U.dateLabel(v.at)+'</small></span>'
        + '<button class="btn btn-soft sm" onclick="Project.restoreVersion(\''+v.id+'\')">Phục hồi</button></div>';
    }).join('');
  }

  function restoreVersion(id){
    var versions=LS.get('versions', []);
    var v=null;
    versions.forEach(function(x){ if(x.id===id) v=x; });
    if(!v) return;
    Poster.setDoc(JSON.parse(JSON.stringify(v.doc)));
    closeModal('modalVersions');
    go('poster');
    toast('Đã phục hồi phiên bản');
  }

  function openVersions(){ renderVersions(); openModal('modalVersions'); }

  function exportJSON(){
    var doc=Poster.doc;
    if(!doc){ toast('Chưa có thiết kế'); return; }
    U.dl('canvio-du-an-'+doc.id+'.json', JSON.stringify({app:'CANVIO Studio 16.2', doc:doc}, null, 2), 'application/json');
    toast('Đã tải tệp dự án');
  }

  function importJSON(ev){
    var f=ev.target.files && ev.target.files[0];
    if(!f) return;
    var r=new FileReader();
    r.onload=function(){
      try{
        var data=JSON.parse(r.result);
        var doc = data.doc || data;
        if(!doc || !doc.layers) throw new Error('bad');
        doc.id = U.uid('doc');
        Poster.setDoc(doc);
        go('poster');
        toast('Đã nhập dự án');
      }catch(e){ toast('Tệp không hợp lệ'); }
    };
    r.readAsText(f);
    ev.target.value='';
  }

  function exportAll(){
    var payload = {
      app:'CANVIO Studio 16.2',
      exportedAt: new Date().toISOString(),
      current: Poster.doc,
      projects: LS.get('projects', []),
      versions: LS.get('versions', []),
      brand: LS.get('brand', null),
      notes: LS.get('notes', []),
      activity: LS.get('activity', [])
    };
    U.dl('canvio-toan-bo-'+Date.now()+'.json', JSON.stringify(payload, null, 2), 'application/json');
    toast('Đã xuất toàn bộ dữ liệu');
  }

  function restore(){
    var doc = LS.get('currentDoc', null);
    if(doc && doc.layers) return doc;
    return null;
  }

  return {autosave:autosave, save:save, createVersion:createVersion, renderVersions:renderVersions,
    restoreVersion:restoreVersion, openVersions:openVersions, exportJSON:exportJSON, importJSON:importJSON,
    exportAll:exportAll, restore:restore};
})();

/* ============================================================
   AI STUDIO — prompt → product
   ============================================================ */
var Studio = (function(){
  var lastResults = [];
  var lastKind = 'poster';

  function parsedHeadline(prompt){
    var m=String(prompt||'').match(/["“]([^"”]{3,60})["”]/);
    if(m) return m[1].toUpperCase();
    var caps=String(prompt||'').match(/\b([A-ZĐ][A-ZĐ0-9'’\s!.]{4,40})\b/);
    return caps? caps[1].trim() : null;
  }

  function currentRatio(){
    var sel=document.getElementById('studioRatio');
    return sel? sel.value : 'portrait';
  }

  function run(prompt, opts){
    opts = opts||{};
    var parse = Designer.parse(prompt);
    var kind = opts.kind || parse.kind || 'poster';
    var ratio = opts.ratio || currentRatio();
    var size = RATIOS[ratio] || RATIOS.portrait;
    var count = opts.count || parseInt((document.getElementById('studioCount')||{}).value||'3',10);
    var selEl = document.getElementById('studioPalette');
    var palId = opts.palette || (selEl && selEl.value? selEl.value : parse.palette);
    if(palId==='auto'||!palId) palId = null;

    var industry = parse.industry;
    var style = parse.style;
    var results=[];
    var variants = [0,1,2,3,4].slice(0, Math.max(2,Math.min(5,count)));
    variants.forEach(function(v){
      var doc = Designer.build({
        w:size.w, h:size.h, ratio:ratio, palette: palId, style: style, industry: industry,
        variant: v, name: labelFor(industry, style, v), brandName: brandName(prompt), kind: kind,
        raw: prompt, headline: parsedHeadline(prompt)
      });
      results.push(doc);
    });
    lastResults = results;
    lastKind = kind;
    return {results:results, parse:parse, kind:kind, ratio:ratio};
  }

  function brandName(prompt){
    var m = String(prompt||'').match(/"[^"]{2,24}"|“[^”]{2,24}”/);
    if(m) return m[0].replace(/["“”]/g,'').toUpperCase();
    return 'NORTHSTAR';
  }

  function labelFor(ind, style, v){
    var i=(Designer.INDUSTRY[ind]||{}).label||'Thiết kế';
    var names=['Bố cục biên tập','Bố cục chia khối','Bố cục căn giữa','Bố cục dải màu','Bố cục góc động'];
    return i+' · '+names[v%5];
  }

  /* ---------- pipeline animation ---------- */
  var PIPE_STEPS = [
    ['Đọc câu lệnh','Nhận diện ngành, phong cách, cảm giác'],
    ['Chọn bảng màu','Đối chiếu bộ màu phù hợp ngành'],
    ['Viết nội dung','Tiêu đề, mô tả, nút kêu gọi'],
    ['Dựng bố cục','Căn lề, phân cấp, khoảng thở'],
    ['Kiểm tra & đóng gói','Tương phản, tuân thủ, bản xuất']
  ];

  function paintPipeline(state){
    var box=document.getElementById('pipeline');
    if(!box) return;
    box.innerHTML = PIPE_STEPS.map(function(s,i){
      var cls = state==='idle'? '' : (i<state? 'ok' : (i===state? 'run':''));
      var icon = i<state? '✓' : (i===state? '◌':'○');
      var color = i<state? 'var(--mint)' : (i===state? 'var(--brand)':'var(--ink-3)');
      return '<div class="listRow" style="gap:9px"><span style="color:'+color+';font-weight:800;width:14px">'+icon+'</span>'
        + '<span style="flex:1"><b style="font-size:11.4px">'+s[0]+'</b><br><small class="tiny">'+s[1]+'</small></span></div>';
    }).join('');
    var st=document.getElementById('pipeState');
    if(st) st.textContent = state==='idle'? 'Sẵn sàng' : (state>=PIPE_STEPS.length? 'Hoàn tất' : 'Đang chạy…');
  }

  var timers=[];
  function animate(done){
    timers.forEach(clearTimeout); timers=[];
    paintPipeline(0);
    PIPE_STEPS.forEach(function(_,i){
      timers.push(setTimeout(function(){ paintPipeline(i+1); }, 260*(i+1)));
    });
    timers.push(setTimeout(function(){ if(done) done(); }, 260*(PIPE_STEPS.length+1)));
  }

  /* ---------- public generate ---------- */
  function generate(source){
    var ta = document.getElementById(source==='home'? 'homePrompt':'studioPrompt');
    var prompt = ta? ta.value.trim() : '';
    if(!prompt){ toast('Nhập mô tả trước nhé'); if(ta) ta.focus(); return; }

    var kind = source==='home'? Home.kind() : ((document.getElementById('studioKind')||{}).value||'poster');
    var parsed = Designer.parse(prompt);
    var ratio = source==='home'? (parsed.ratio||'portrait') : currentRatio();

    go('studio');
    var out = run(prompt, {kind:kind, ratio:ratio});
    paintRead(out.parse, prompt);
    animate(function(){
      renderResults(out.results, out.parse);
      Board.log('Tạo '+out.results.length+' phương án từ câu lệnh');
      toast('Đã tạo '+out.results.length+' phương án');
    });
    var box=document.getElementById('studioResults');
    if(box) box.innerHTML='<div class="emptyState" style="grid-column:1/-1"><div class="ei">◌</div><b>Đang dựng bố cục…</b><div class="tiny">Quá trình gồm 5 bước, khoảng 1–2 giây.</div></div>';
  }

  function renderResults(docs, parse){
    var box=document.getElementById('studioResults');
    if(!box) return;
    box.innerHTML = docs.map(function(d, i){
      return '<div class="pvCard"><div class="pvThumb" style="height:220px">'+Preview.thumb(d, 240, 220)+'</div>'
        + '<div class="pvBody"><div><b>'+U.esc(d.name)+'</b><small>'+U.esc(palName(d.palette))+' · '+d.w+'×'+d.h+'</small></div>'
        + '<div class="row" style="gap:4px">'
        + '<button class="btn btn-ghost sm" onclick="Studio.inspect('+i+')">Xem</button>'
        + '<button class="btn btn-soft sm" onclick="Studio.accept('+i+')">Dùng</button>'
        + '</div></div></div>';
    }).join('');
    var meta=document.getElementById('resultMeta');
    if(meta) meta.textContent = docs.length+' phương án · '+((Designer.INDUSTRY[parse.industry]||{}).label||'')+' · '+docs[0].w+'×'+docs[0].h;
  }

  function palName(id){
    var p=Designer.paletteById(id);
    return p? p.name : id;
  }

  function accept(i){
    var d=lastResults[i];
    if(!d) return;
    var clone=JSON.parse(JSON.stringify(d));
    clone.id=U.uid('doc');
    clone.layers.forEach(function(L){ L.id=U.uid('ly'); });
    Poster.setDoc(clone);
    Board.log('Áp dụng phương án “'+d.name+'”');
    if(lastKind==='web'){
      Web.fromPoster();
      go('web');
      toast('Đã chuyển nội dung sang Website Builder');
      return;
    }
    if(lastKind==='set'){
      go('responsive');
      toast('Đã tạo bộ đa kích thước từ thiết kế này');
      return;
    }
    go('poster');
    toast('Đã mở trong Poster Studio');
  }

  function inspect(i){
    var d=lastResults[i];
    if(!d) return;
    openLab(d.name, d.w+'×'+d.h+' · '+palName(d.palette), layerSummary(d));
  }

  function layerSummary(d){
    return d.layers.map(function(L){
      var t={text:'Chữ',badge:'Nhãn',rect:'Khối',ellipse:'Tròn',line:'Đường',image:'Ảnh'}[L.type]||L.type;
      var txt = (L.type==='text'||L.type==='badge')? ' — “'+U.esc((L.text||'').slice(0,34))+'”':'';
      return '<div class="spread" style="padding:4px 0;border-bottom:1px dashed var(--line)"><span><b>'+t+'</b>'+txt+'</span><small class="tiny">'+Math.round(L.x)+','+Math.round(L.y)+' · '+Math.round(L.w)+'×'+Math.round(L.h)+'</small></div>';
    }).join('');
  }

  function paintRead(parse, prompt){
    var box=document.getElementById('promptRead');
    if(!box) return;
    var ind=(Designer.INDUSTRY[parse.industry]||{}).label||parse.industry;
    var st=STYLE_LIB.filter(function(s){return s.id===parse.style;})[0];
    var pal = parse.palette==='auto'? ('Tự chọn theo ngành — '+palName(Designer.autoPalette(parse.industry).id)) : palName(parse.palette);
    box.innerHTML =
      '<div class="spread"><span>Ngành</span><b>'+U.esc(ind)+'</b></div>'
      + '<div class="spread"><span>Phong cách</span><b>'+U.esc(st?st.name:parse.style)+'</b></div>'

  function guard(){
    var b=read();
    var doc=Poster.doc;
    var rows=[];
    var ratio=U.contrast(b.ink,b.bg);
    rows.push(ratio>=4.5? {ok:true,msg:'Tương phản chữ/nền đạt '+ratio.toFixed(2)+':1'} : {ok:false,msg:'Tương phản chỉ '+ratio.toFixed(2)+':1 (cần ≥ 4.5)'});
    if(doc){
      var pal=Designer.paletteById(doc.palette);
      var brandColors=[b.primary,b.secondary,b.accent].map(function(x){return (x||'').toLowerCase();});
      var texts=doc.layers.filter(function(L){return L.type==='text';});
      var offBrand=texts.filter(function(L){ return L.color && L.color[0]==='#' && brandColors.indexOf(L.color.toLowerCase())===-1 && L.color.toLowerCase()!==b.ink.toLowerCase(); });
      rows.push(offBrand.length===0? {ok:true,msg:'Màu chữ chủ đạo nằm trong bộ thương hiệu'} : {ok:true,msg:offBrand.length+' lớp dùng màu ngoài bảng (có thể là chủ ý)'});
      var ctas=doc.layers.filter(function(L){return L.type==='badge';});
      rows.push(ctas.length? {ok:true,msg:'Có '+ctas.length+' nút kêu gọi hành động'} : {ok:false,msg:'Thiếu nút kêu gọi hành động'});
      var sizes={};
      texts.forEach(function(L){ sizes[Math.round(L.size||0)]=1; });
      var unique=Object.keys(sizes).length;
      rows.push(unique<=4? {ok:true,msg:unique+' cấp cỡ chữ (khuyến nghị ≤ 4)'} : {ok:false,msg:unique+' cấp cỡ chữ — nên gộp lại'});
      var margin=Math.round(Math.min(doc.w,doc.h)*0.04);
      var tight=doc.layers.filter(function(L){ return L.type!=='rect' && L.type!=='image' && (L.x<margin*0.4 || L.y<margin*0.4); });
      rows.push(tight.length? {ok:false,msg:tight.length+' lớp sát mép — nên tăng lề an toàn'} : {ok:true,msg:'Lề an toàn hợp lệ'});
      var palOk = pal.c.primary.toLowerCase()===b.primary.toLowerCase() || pal.c.accent.toLowerCase()===b.accent.toLowerCase();
      rows.push(palOk? {ok:true,msg:'Bảng màu thiết kế khớp thương hiệu'} : {ok:true,msg:'Bảng màu thiết kế khác thương hiệu — cân nhắc đồng bộ'});
    } else {
      rows.push({ok:false,msg:'Chưa có thiết kế để đối chiếu'});
    }
    var box=document.getElementById('guardResults');
    if(box){
      box.innerHTML = rows.map(function(r){
        return '<div class="advice '+(r.ok?'ok':'fix')+'"><span class="ico">'+(r.ok?'✓':'✕')+'</span><span>'+r.msg+'</span></div>';
      }).join('');
    }
    Board.log('Chạy kiểm tra thương hiệu');
  }

      var e=document.getElementById(id);
      if(e) e.addEventListener('input', preview);
    });
  }
  return {init:init, paint:paint, save:save, preview:preview, guard:guard, syncFromDesign:syncFromDesign, read:read};
})();

var Assets = (function(){
  function list(){ return LS.get('assets', []); }
  function render(){
    var grid=document.getElementById('assetGrid');
    if(!grid) return;
    var q=(document.getElementById('assetSearch')||{}).value||'';
    var f=(document.getElementById('assetFilter')||{}).value||'all';

    });
    var meta=document.getElementById('assetMeta');
    if(meta) meta.textContent = items.length+' tài nguyên · '+U.fmtBytes(list().reduce(function(s,a){return s+(a.size||0);},0));
    if(!items.length){
      grid.innerHTML='<div class="emptyState" style="grid-column:1/-1"><div class="ei">▦</div><b>Chưa có tài nguyên</b><div class="tiny">Kéo ảnh vào ô phía trên để bắt đầu.</div></div>';
      Board.refresh();
      return;
    }
    grid.innerHTML = items.map(function(a){
      return '<div class="assetCard"><div class="assetThumb">'
        + (a.data? '<img src="'+a.data+'" alt="'+U.esc(a.name)+'" loading="lazy">':'<span class="tiny">Không xem trước</span>')
        + '</div><div class="assetInfo"><b title="'+U.esc(a.name)+'">'+U.esc(a.name)+'</b><small>'+U.esc(a.cat||'')+' · '+U.fmtBytes(a.size||0)+'</small>'
        + '<div class="row" style="gap:4px;margin-top:6px">'
        + '<button class="btn btn-soft sm" onclick="Assets.use(\''+a.id+'\')">Chèn</button>'
        + '<button class="btn btn-danger sm" onclick="Assets.remove(\''+a.id+'\')">Xoá</button>'
        + '</div></div></div>';
    }).join('');
    Board.refresh();
  }
  function catOf(name, mime){
    var n=String(name||'').toLowerCase();
    if(n.indexOf('logo')>-1) return 'logo';
    if(n.indexOf('icon')>-1 || n.indexOf('bieu-tuong')>-1) return 'icon';
    if(mime==='image/svg+xml') return 'svg';
    return 'image';
  }
  function readFile(f){
    return new Promise(function(res){
      var r=new FileReader();
      r.onload=function(){ res(r.result); };
      r.onerror=function(){ res(null); };
      r.readAsDataURL(f);
    });
  }
  function upload(ev){
    var files=Array.prototype.slice.call(ev.target.files||[]);
    if(!files.length) return;
    var arr=list();
    var pending=files.length, added=0;
    files.forEach(function(f){
      if(f.size>5*1024*1024){ toast('“'+f.name+'” quá 5 MB'); pending--; return; }
      readFile(f).then(function(data){
        if(data){
          arr.unshift({id:U.uid('as'), name:f.name, mime:f.type, size:f.size, cat:catOf(f.name,f.type), data:data, at:new Date().toISOString()});
          added++;
        }
        pending--;
        if(pending===0){
          LS.set('assets', arr.slice(0,40));
          render();
          Board.log('Tải lên '+added+' tài nguyên');
          toast('Đã thêm '+added+' tài nguyên');
        }
      });
    });
    ev.target.value='';
  }
  function use(id){
    var a=null;
    list().forEach(function(x){ if(x.id===id) a=x; });
    if(!a) return;
    var doc=Poster.doc;
    if(!doc){ toast('Chưa có thiết kế'); return; }
    var M=Math.round(Math.min(doc.w,doc.h)*0.1);
    var w=Math.round(doc.w*0.5);
    var L=Designer.mkLayer({type:'image', src:a.data, x:M, y:M, w:w, h:Math.round(w*0.66)});
    doc.layers.push(L);
    Poster.render(); Poster.select(L.id); Poster.history.push(); Project.autosave();
    go('poster');
    toast('Đã chèn “'+a.name+'”');
  }
  function remove(id){
    LS.set('assets', list().filter(function(a){ return a.id!==id; }));
    render(); toast('Đã xoá tài nguyên');
  }
  function backup(){
    U.dl('canvio-tai-nguyen-'+Date.now()+'.json', JSON.stringify({app:'CANVIO Studio 16.2', assets:list()}, null, 2), 'application/json');
    toast('Đã tải bản sao lưu');
  }
  function restore(ev){
    var f=ev.target.files && ev.target.files[0];
    if(!f) return;
    var r=new FileReader();
    r.onload=function(){
      try{
        var d=JSON.parse(r.result);
        var items=(d.assets||[]).filter(function(a){ return a && a.id && a.data; });
        if(!items.length) throw new Error('empty');
        LS.set('assets', items.concat(list()).slice(0,40));
        render(); toast('Đã phục hồi '+items.length+' tài nguyên');
      }catch(e){ toast('Tệp sao lưu không hợp lệ'); }
    };
    r.readAsText(f);
    ev.target.value='';
  }
  function init(){
    var dz=document.getElementById('dropzone');
    if(dz){
      ['dragenter','dragover'].forEach(function(ev){
        dz.addEventListener(ev, function(e){ e.preventDefault(); dz.classList.add('drag'); });
      });
      ['dragleave','drop'].forEach(function(ev){
        dz.addEventListener(ev, function(e){ e.preventDefault(); dz.classList.remove('drag'); });
      });
      dz.addEventListener('drop', function(e){
        if(e.dataTransfer && e.dataTransfer.files.length) upload({target:{files:e.dataTransfer.files, value:''}});
      });
    }
  }
  return {init:init, render:render, upload:upload, use:use, remove:remove, backup:backup, restore:restore, list:list, addBlob:function(file){return new Promise(function(resolve,reject){var r=new FileReader();r.onload=function(){var arr=list();arr.unshift({id:U.uid('as'),name:file.name,mime:file.type,size:file.size,cat:catOf(file.name,file.type),data:r.result,at:new Date().toISOString()});LS.set('assets',arr.slice(0,40));render();resolve()};r.onerror=function(){reject(r.error||new Error("CANVIO: không tải/đọc được tài nguyên"))};r.readAsDataURL(file)})}};
})();

var Responsive = (function(){
  function render(){
    var grid=document.getElementById('respGrid');
    if(!grid) return;
    var src=Poster.doc;
    if(!src){ grid.innerHTML='<div class="emptyState"><div class="ei">▥</div><b>Chưa có thiết kế</b><div class="tiny">Mở Poster Studio để tạo trước.</div></div>'; return; }
    var meta=document.getElementById('respMeta');
    if(meta) meta.textContent='Nguồn: “'+(src.name||'Thiết kế')+'” · '+src.w+'×'+src.h+' · '+SIZES.length+' kích thước';
    grid.innerHTML = SIZES.map(function(s, i){
      var doc=adapt(src, s);
      var boxH = s.h>s.w? 210 : 150;
      return '<div class="pvCard" data-size-id="'+s.id+'"><div class="pvThumb" style="height:'+boxH+'px">'+Preview.thumb(doc, 240, boxH)+'</div>'
        + '<div class="pvBody"><div><b>'+U.esc(s.name)+'</b><small>'+s.w+'×'+s.h+' · '+U.esc(s.group)+'</small></div>'
        + '<button class="btn btn-soft sm" onclick="Responsive.exportOne('+i+')">Xuất</button></div></div>';
    }).join('');
  }
  function adapt(src, size){
    var doc=JSON.parse(JSON.stringify(src));
    var sx=size.w/src.w, sy=size.h/src.h;
    var s=Math.min(sx,sy);
    doc.w=size.w; doc.h=size.h; doc.ratio=size.ratio;
    doc.sizeName=size.name;
    doc.layers.forEach(function(L){
      var isBg = (L.type==='rect'||L.type==='image') && L.x<=1 && L.y<=1 && L.w>=src.w*0.98 && L.h>=src.h*0.98;
      if(isBg){ L.x=0; L.y=0; L.w=size.w; L.h=size.h; return; }
      L.x=Math.round(L.x*sx); L.y=Math.round(L.y*sy);
      L.w=Math.round(L.w*sx); L.h=Math.round(L.h*sy);
      if(L.size) L.size=Math.max(5, Math.round(L.size*s));
      if(L.spacing) L.spacing=Math.round(L.spacing*s*10)/10;
      if(L.radius && L.radius!==999) L.radius=Math.round(L.radius*s);
      L.x=U.clamp(L.x, -L.w*0.3, size.w-L.w*0.4);
      L.y=U.clamp(L.y, -L.h*0.3, size.h-L.h*0.4);
    });
    return doc;
  }
  function exportOne(i){
    var s=SIZES[i];
    if(!s || !Poster.doc) return;
    var doc=adapt(Poster.doc, s);
    Export.renderDoc(doc, 'png', 'canvio-'+s.id);
  }
  function exportAll(){
    if(!Poster.doc){ toast('Chưa có thiết kế'); return; }
    var payload={app:'CANVIO Studio 16.2', base:{name:Poster.doc.name}, sizes:SIZES.map(function(s){ return {id:s.id,name:s.name,w:s.w,h:s.h,doc:adapt(Poster.doc,s)}; })};
    U.dl('canvio-bo-kich-thuoc.json', JSON.stringify(payload,null,2),'application/json');
    toast('Đã xuất bộ kích thước');
  }
  function exportZipish(selectedIds){
    if(!Poster.doc){ toast('Chưa có thiết kế'); return; }
    var selected = Array.isArray(selectedIds) ? SIZES.filter(function(s){ return selectedIds.indexOf(s.id) >= 0; }) : SIZES.slice();
    if(!selected.length){ toast('Hãy chọn ít nhất một kích thước để xuất.'); return false; }
    if(!window.CVZip){ selected.forEach(function(s,k){ var idx=SIZES.indexOf(s); setTimeout(function(){ exportOne(idx); }, k*650); }); return false; }
    toast('Đang kết xuất '+selected.length+' / '+SIZES.length+' kích thước PNG…');
    var files=[], manifest=[], i=0;
    (function next(){
      if(i>=selected.length){
        files.push({name:'manifest.json',data:JSON.stringify(manifest,null,2)});
        CVZip.build(files).then(function(z){ U.dl('canvio-bo-kich-thuoc.zip', z, 'application/zip'); toast('Đã tải ZIP gồm '+manifest.length+' ảnh PNG đã chọn'); }).catch(function(e){ toast('Không đóng gói được ZIP: '+String(e&&e.message||e).slice(0,90)); });
        return;
      }
      var s=selected[i++], doc=adapt(Poster.doc,s);
      Export.toBlob(doc,1,function(b){
        if(b){ files.push({name:'canvio-'+s.id+'-'+s.w+'x'+s.h+'.png',data:b}); manifest.push({id:s.id,name:s.name,w:s.w,h:s.h}); }
        setTimeout(next,0);
      });
    })();
    return true;
  }
  return {render:render, adapt:adapt, getSizes:function(){return SIZES.slice();}, exportOne:exportOne, exportAll:exportAll, exportZipish:exportZipish};
})();

var Motion = (function(){
  var current = 'fade-up';

  function render(){
    var list=document.getElementById('motionList');
    if(list && !list.dataset.done){
      list.dataset.done='1';
      list.innerHTML = MOTIONS.map(function(m){
        return '<button class="chip'+(m.id===current?' on':'')+'" style="'+(m.id===current?'border-color:#b9bfff;color:var(--brand-ink);background:#f7f8ff':'')+'" data-m="'+m.id+'">'+m.name+'</button>';
      }).join('');
      var bs=list.querySelectorAll('button');
      for(var i=0;i<bs.length;i++){
        bs[i].addEventListener('click', function(){
          current=this.getAttribute('data-m');
          for(var j=0;j<bs.length;j++){
            bs[j].classList.remove('on');
            bs[j].style.borderColor=''; bs[j].style.color=''; bs[j].style.background='';
          }
          this.style.borderColor='#b9bfff'; this.style.color='var(--brand-ink)'; this.style.background='#f7f8ff';
          play();
        });
      }
    }
    play();
  }

  function spec(){
    for(var i=0;i<MOTIONS.length;i++) if(MOTIONS[i].id===current) return MOTIONS[i];
    return MOTIONS[0];
  }

  function play(){
    var box=document.getElementById('motionStage');
    if(!box) return;
    var doc=Poster.doc;
    var dur=parseInt((document.getElementById('motionDur')||{}).value||700,10);
    var stag=parseInt((document.getElementById('motionStag')||{}).value||60,10);
    var ease=(document.getElementById('motionEase')||{}).value||'cubic-bezier(.2,.8,.3,1)';
    var dv=document.getElementById('durVal'); if(dv) dv.textContent=dur+'ms';
    var sv=document.getElementById('stagVal'); if(sv) sv.textContent=stag+'ms';
    var nm=document.getElementById('motionName'); if(nm) nm.textContent=current;

    if(!doc){
      box.innerHTML='<div class="emptyState"><div class="ei">▶</div><b>Chưa có thiết kế</b><div class="tiny">Mở Poster Studio để tạo trước.</div></div>';
      return;
    }
    var s=spec();
    var z=Math.min(300/doc.w, 340/doc.h, 1);
    box.innerHTML='';
    box.innerHTML = '<div style="width:'+Math.round(doc.w*z)+'px;height:'+Math.round(doc.h*z)+'px;overflow:hidden;position:relative;box-shadow:0 18px 40px rgba(16,32,60,.16);background:#fff;border-radius:6px">'

      + Styler.all(doc) + '</div></div>';
    var inner=document.getElementById('motionInner');
    if(!inner) return;
    var nodes=Array.prototype.slice.call(inner.children);
    nodes.forEach(function(n, i){
      n.style.transition='opacity '+dur+'ms '+ease+', transform '+dur+'ms '+ease+' '+(i*stag)+'ms, filter '+dur+'ms';
      if(s.blur) n.style.filter=s.blur;
      if(s.clip) n.style.clipPath='inset(100% 0 0 0)';
      n.style.opacity=s.opacity;
      if(s.from!=='none') n.style.transform=(n.style.transform||'')+' '+s.from;
    });
    setTimeout(function(){
      nodes.forEach(function(n){
        n.style.opacity=1;
        n.style.transform=(n.style.transform||'').replace(s.from,'');
        n.style.filter='';
        n.style.clipPath='';
      });
    }, 60);
  }

  function css(){
    var doc=Poster.doc;
    var dur=parseInt((document.getElementById('motionDur')||{}).value||700,10);
    var stag=parseInt((document.getElementById('motionStag')||{}).value||60,10);
    var ease=(document.getElementById('motionEase')||{}).value||'cubic-bezier(.2,.8,.3,1)';
    var s=spec();
    var lines=[];
    lines.push('/* CANVIO Studio 16.2 — chuyển động: '+s.name+' */');

    lines.push('.cv-anim > *{opacity:'+(s.opacity||1)+';'+(s.from!=='none'?'transform:'+s.from+';':'')+(s.blur?'filter:'+s.blur+';':'')+'}');

/* ============================================================
   INSIGHT — heuristic design scoring
   ============================================================ */
var Insight = (function(){
  var last=null;

  function metrics(doc){
    var L=doc? doc.layers : [];
    var texts=L.filter(function(x){return x.type==='text';});
    var ctas=L.filter(function(x){return x.type==='badge';});
    var pal=Designer.paletteById(doc? doc.palette : 'indigo');
    var out=[];

    var contrast=U.contrast(pal.c.ink, pal.c.bg1);
    out.push({k:'Tương phản', v:U.clamp(Math.round(contrast/7*100),0,100), raw:contrast.toFixed(2)+':1'});

    var sizes=texts.map(function(t){ return t.size||12; }).sort(function(a,b){return b-a;});
    var hierarchy = 30;
    if(sizes.length>=2 && sizes[0]>0){
      var r=sizes[0]/sizes[sizes.length-1];
      hierarchy = U.clamp(Math.round((r/4)*100), 20, 100);
    } else if(sizes.length===1) hierarchy=55;
    out.push({k:'Phân cấp chữ', v:hierarchy, raw:sizes.length+' cấp cỡ chữ'});

    var fill=0;
    L.forEach(function(x){
      var isBg = (x.type==='rect'||x.type==='image') && x.x<=1 && x.y<=1 && x.w>=doc.w*0.97 && x.h>=doc.h*0.97;
      if(isBg) return;
      fill += (x.w*x.h)/(doc.w*doc.h);
    });
    var density = U.clamp(Math.round((1-Math.abs(fill-0.34)/0.5)*100), 5, 100);
    out.push({k:'Mật độ', v:density, raw:Math.round(fill*100)+'% diện tích'});

    var margin = Math.round(Math.min(doc.w,doc.h)*0.072);
    var offCount = L.filter(function(x){ return x.type!=='rect' && x.type!=='image' && (x.x<margin*0.35 || x.y<margin*0.35 || x.x+x.w>doc.w-margin*0.35); }).length;
    var align = U.clamp(100-offCount*22, 10, 100);
    out.push({k:'Căn lề & lề an toàn', v:align, raw:offCount? offCount+' lớp sát mép':'đạt'});

    var cta = ctas.length===1? 100 : ctas.length===0? 15 : Math.max(45, 100-(ctas.length-1)*20);
    out.push({k:'Kêu gọi hành động', v:cta, raw:ctas.length? ctas.length+' nút':'thiếu nút'});

    var bal=100;
    var left=0, right=0;
    L.forEach(function(x){
      var isBg = (x.type==='rect'||x.type==='image') && x.x<=1 && x.y<=1 && x.w>=doc.w*0.97 && x.h>=doc.h*0.97;
      if(isBg) return;
      var c=x.x+x.w/2; if(c<doc.w/2) left+=x.w*x.h; else right+=x.w*x.h;
    });
    if(left+right>0){ var skew=Math.abs(left-right)/(left+right); bal=U.clamp(Math.round(100-skew*90), 20, 100); }
    out.push({k:'Cân bằng', v:bal, raw:(bal>75?'cân đối':'lệch '+(100-bal)+'%')});

    return out;
  }

  function run(){
    var doc=Poster.doc;
    if(!doc){ toast('Chưa có thiết kế để đánh giá'); return; }
    var m=metrics(doc);
    var total=Math.round(m.reduce(function(s,x){return s+x.v;},0)/m.length);
    last={doc:doc, metrics:m, total:total};
    paint();
    Board.log('Chấm điểm thiết kế: '+total+'/100');
    toast('Điểm tổng: '+total+'/100');
  }

  function paint(){
    var doc=Poster.doc;
    var m = last && last.doc && doc && last.doc.id===doc.id ? last.metrics : (doc? metrics(doc): []);
    var total = m.length? Math.round(m.reduce(function(s,x){return s+x.v;},0)/m.length) : 0;

    var ring=document.getElementById('scoreRing');
    if(ring){
      var R=56, C=2*Math.PI*R;
      var dash = m.length? (C*total/100) : 0;
      ring.innerHTML =
        '<svg width="132" height="132" viewBox="0 0 132 132">'
        + '<circle cx="66" cy="66" r="'+R+'" fill="none" stroke="#eef2fa" stroke-width="11"/>'
        + '<circle cx="66" cy="66" r="'+R+'" fill="none" stroke="url(#cvg)" stroke-width="11" stroke-linecap="round" stroke-dasharray="'+dash.toFixed(1)+' '+C.toFixed(1)+'"/>'
        + '<defs><linearGradient id="cvg" x1="0" y1="0" x2="1" y2="1"><stop offset="0" stop-color="#6366f1"/><stop offset="1" stop-color="#0ea5e9"/></linearGradient></defs>'
        + '</svg>'
        + '<div class="val"><div><strong>'+(m.length? total:'—')+'</strong><small>/ 100</small></div></div>';
    }
    var note=document.getElementById('scoreNote');
    if(note) note.textContent = m.length? (total>=85? 'Rất tốt — sẵn sàng phát hành': total>=70? 'Khá tốt — còn vài điểm cần chỉnh': 'Cần cải thiện trước khi xuất') : 'Chạy đánh giá để chấm điểm';

    var list=document.getElementById('metricList');
    if(list){
      if(!m.length){ list.innerHTML='<div class="tiny">Chạy đánh giá để xem chi tiết.</div>'; }
      else {
        list.innerHTML = m.map(function(x){
          var color = x.v>=80? 'var(--mint)' : x.v>=60? 'var(--amber)' : 'var(--rose)';
          return '<div class="metric"><span class="mname">'+x.k+'</span><div class="mbar"><i style="width:'+x.v+'%;background:'+color+'"></i></div><span class="mval">'+x.v+'</span></div>';
        }).join('');
      }
    }
    var meta=document.getElementById('insightMeta');
    if(meta) meta.textContent = m.length? 'Chấm theo 6 tiêu chí heuristic' : '';

    heatmap();
    advice(m);
  }

  function heatmap(){
    var grid=document.getElementById('heatmap');
    if(!grid) return;
    var doc=Poster.doc;
    var cols=14, rows=10;
    grid.style.gridTemplateColumns='repeat('+cols+',1fr)';
    var cells='';
    var hotspots={x:0.5,y:0.36,w:0.4,h:0.22,intensity:1};
    if(doc){
      var head=doc.layers.filter(function(L){return L.type==='text';})[0];
      if(head) hotspots={x:(head.x+head.w/2)/doc.w, y:(head.y+head.h/2)/doc.h, w:head.w/doc.w, h:head.h/doc.h, intensity:1};
    }
    for(var r=0;r<rows;r++){
      for(var c=0;c<cols;c++){
        var nx=(c+0.5)/cols, ny=(r+0.5)/rows;
        var d=Math.sqrt(Math.pow((nx-hotspots.x)/ (hotspots.w+0.25),2)+Math.pow((ny-hotspots.y)/(hotspots.h+0.25),2));
        var v=U.clamp(1-d*0.95, 0, 1);
        v=Math.pow(v,1.4);
        cells += '<i style="background:rgba(79,70,229,'+(0.06+v*0.78).toFixed(2)+')"></i>';
      }
    }
    grid.innerHTML=cells;
  }

  function advice(m){
    var box=document.getElementById('adviceList');
    var cnt=document.getElementById('adviceCount');
    if(!box) return;
    var rows=[];
    m.forEach(function(x){
      if(x.v>=85) rows.push({t:'ok', msg:'<b>'+x.k+'</b> — '+x.raw+' · đạt tốt'});
      else if(x.v>=60) rows.push({t:'warn', msg:'<b>'+x.k+'</b> — '+x.raw+' · có thể cải thiện'});
      else rows.push({t:'fix', msg:'<b>'+x.k+'</b> — '+x.raw+' · nên chỉnh'});
    });
    if(!rows.length) rows.push({t:'warn', msg:'Chưa có dữ liệu để gợi ý.'});
    box.innerHTML = rows.map(function(r){
      var icon = r.t==='ok'? '✓' : r.t==='warn'? '!' : '✕';
      return '<div class="advice '+r.t+'"><span class="ico">'+icon+'</span><span>'+r.msg+'</span></div>';
    }).join('');
    if(cnt) cnt.textContent = rows.length+' gợi ý';
  }

  function autoFix(){
    var doc=Poster.doc;
    if(!doc){ toast('Chưa có thiết kế'); return; }
    var pal=Designer.paletteById(doc.palette);
    var margin=Math.round(Math.min(doc.w,doc.h)*0.072);
    var fixes=0;
    doc.layers.forEach(function(L){
      if(L.type==='text'||L.type==='badge'){
        if((L.color||'').toLowerCase()===pal.c.bg1.toLowerCase()){ L.color=pal.c.ink; fixes++; }
        if(L.x<margin && L.x>-margin*0.5){ L.x=margin; fixes++; }
        if(L.y<margin && L.y>-margin*0.5){ L.y=margin; fixes++; }
        if(L.x+L.w>doc.w-margin*0.5){ L.x=Math.max(0, doc.w-margin-L.w); fixes++; }
      }
    });
    var sizes=doc.layers.filter(function(L){return L.type==='text';}).map(function(L){return L.size;}).sort(function(a,b){return b-a;});
    if(sizes.length && sizes[0]/sizes[sizes.length-1]<1.8){
      var texts=doc.layers.filter(function(L){return L.type==='text';});
      texts[0].size=Math.round(sizes[0]*1.45);
      fixes++;
    }
    if(fixes===0){ toast('Không phát hiện điểm cần sửa tự động'); return; }
    Poster.render(); Poster.history.push(); Project.autosave();
    last=null; paint();
    toast('Đã sửa '+fixes+' điểm');
  }

  function init(){ paint(); }
  return {init:init, run:run, paint:paint, autoFix:autoFix, metrics:metrics};
})();

/* ============================================================
   EXPORT — one SVG renderer, several outputs
   ============================================================ */
var Export = (function(){

  var measureCtx = null;
  function measure(){
    if(measureCtx) return measureCtx;
    var c=document.createElement('canvas');
    measureCtx=c.getContext('2d');
    return measureCtx;
  }

  function fontStr(L){
    return (L.weight||700)+' '+(L.size||14)+'px '+FONTS[L.font||'sans'];
  }

  function wrap(L){
    var ctx=measure();
    ctx.font=fontStr(L);
    var maxW=(L.w||200);
    var raw=String(L.text||'').split('\n');
    var lines=[];
    raw.forEach(function(par){
      var words=par.split(/\s+/).filter(function(w){ return w.length; });
      if(!words.length){ lines.push(''); return; }
      var line='';
      words.forEach(function(w){
        var test = line? line+' '+w : w;
        if(ctx.measureText(test).width > maxW && line){ lines.push(line); line=w; }
        else line=test;
      });
      if(line) lines.push(line);
    });
    return lines;
  }

  /* --- gradient parser for the gradients this app generates --- */
  function splitTop(s){
    var out=[], depth=0, cur='';
    for(var i=0;i<s.length;i++){
      var ch=s[i];
      if(ch==='(') depth++;
      if(ch===')') depth--;
      if(ch===',' && depth===0){ out.push(cur); cur=''; continue; }
      cur+=ch;
    }
    if(cur.trim()) out.push(cur);
    return out.map(function(x){ return x.trim(); });
  }

  function colorStop(part){
    var m=part.match(/^(.*?)(?:\s+([\d.]+)%)?$/);
    var col=(m&&m[1]? m[1].trim() : part).trim();
    var pos=m&&m[2]!==undefined? parseFloat(m[2])/100 : null;
    return {col:col, pos:pos};
  }

  function gradDef(id, cssStr, w, h){
    var isRadial = cssStr.indexOf('radial-gradient')===0;
    var inner = cssStr.slice(cssStr.indexOf('(')+1, cssStr.lastIndexOf(')'));
    var parts = splitTop(inner);
    var head = parts.shift()||'';
    var stops = parts.map(colorStop);
    /* normalise positions */
    var known = stops.filter(function(s){ return s.pos!==null; });
    stops.forEach(function(s,i){ if(s.pos===null) s.pos = known.length? (i===0?0:1) : (stops.length>1? i/(stops.length-1):0); });
    for(var i=1;i<stops.length;i++){ if(stops[i].pos<stops[i-1].pos) stops[i].pos=stops[i-1].pos; }

    var def='';
    if(isRadial){
      var at=null, shape='circle';
      var hm=head.match(/at\s+([\d.]+)%\s+([\d.]+)%/);
      if(hm) at={x:parseFloat(hm[1])/100, y:parseFloat(hm[2])/100};
      var cx = at? at.x*w : w/2, cy = at? at.y*h : h/2;
      var r = Math.max(w,h)*0.62;
      def += '<radialGradient id="'+id+'" gradientUnits="userSpaceOnUse" cx="'+cx.toFixed(1)+'" cy="'+cy.toFixed(1)+'" r="'+r.toFixed(1)+'">';
      stops.forEach(function(s){
        def += '<stop offset="'+(s.pos*100).toFixed(1)+'%" stop-color="'+s.col+'"/>';
      });
      def += '</radialGradient>';
    } else {
      var ang=135;
      var am=head.match(/(-?[\d.]+)deg/);
      if(am) ang=parseFloat(am[1]);
      var rad=(ang-90)*Math.PI/180;
      var cx0=w/2 - Math.cos(rad)*w/2, cy0=h/2 - Math.sin(rad)*h/2;
      var cx1=w/2 + Math.cos(rad)*w/2, cy1=h/2 + Math.sin(rad)*h/2;
      def += '<linearGradient id="'+id+'" gradientUnits="userSpaceOnUse" x1="'+cx0.toFixed(1)+'" y1="'+cy0.toFixed(1)+'" x2="'+cx1.toFixed(1)+'" y2="'+cy1.toFixed(1)+'">';
      stops.forEach(function(s){
        def += '<stop offset="'+(s.pos*100).toFixed(1)+'%" stop-color="'+s.col+'"/>';
      });
      def += '</linearGradient>';
    }
    return def;
  }

  function svgString(doc){
    var defs=[], body=[];
    var defCount=0;

    doc.layers.forEach(function(L, idx){
      if(L.visible===false) return;
      if(L.type==='image' && L.src){
        if(window.__cvImg){var __g=window.__cvImg(L); if(__g){ body.push(__g); return; }}
        body.push('<g opacity="'+(L.opacity==null?1:L.opacity)+'"><image href="'+L.src+'" x="'+L.x+'" y="'+L.y+'" width="'+L.w+'" height="'+(L.h||L.w)+'" preserveAspectRatio="xMidYMid slice"'+(L.filter&&/^[\w\s().,%-]+$/.test(L.filter)?' style="filter:'+L.filter+'"':'')+'/></g>');
        return;
      }
      var paint='';
      if(L.gradient){
        var id='g'+(++defCount);
        defs.push(gradDef(id, L.gradient, L.w, L.h||L.w));
        /* gradient coords are layer-local: wrap in translate */
        var isRad = L.gradient.indexOf('radial-gradient')===0;
        paint = '<g transform="translate('+L.x+','+L.y+') rotate('+(L.rotate||0)+' '+L.w/2+' '+(L.h||L.w)/2+')" opacity="'+(L.opacity==null?1:L.opacity)+'">'
          + (L.type==='ellipse'
              ? '<ellipse cx="'+(L.w/2)+'" cy="'+((L.h||L.w)/2)+'" rx="'+(L.w/2)+'" ry="'+((L.h||L.w)/2)+'" fill="url(#'+id+')"/>'
              : '<rect x="0" y="0" width="'+L.w+'" height="'+(L.h||L.w)+'" rx="'+(L.radius&&L.radius!==999? L.radius : 0)+'" fill="url(#'+id+')"/>')
          + '</g>';
        body.push(paint);
        return;
      }

      if(L.type==='rect' || L.type==='ellipse' || L.type==='line'){
        var rx = (L.type==='ellipse')? '50%' : (L.radius===999? Math.min(L.w,(L.h||L.w))/2 : (L.radius||0));
        var trans='transform="rotate('+(L.rotate||0)+' '+(L.x+L.w/2)+' '+(L.y+(L.h||L.w)/2)+')"';
        if(L.type==='ellipse'){
          body.push('<ellipse cx="'+(L.x+L.w/2)+'" cy="'+(L.y+(L.h||L.w)/2)+'" rx="'+(L.w/2)+'" ry="'+((L.h||L.w)/2)+'" fill="'+(L.bg||'none')+'" opacity="'+(L.opacity==null?1:L.opacity)+'" '+trans+'/>');
        } else {
          var r = (L.radius===999)? Math.min(L.w,(L.h||L.w))/2 : (L.radius||0);
          body.push('<rect x="'+L.x+'" y="'+L.y+'" width="'+L.w+'" height="'+(L.h||2)+'" rx="'+r+'" fill="'+(L.bg||'none')+'" opacity="'+(L.opacity==null?1:L.opacity)+'" '+trans+'/>');
        }
        return;
      }

      if(L.type==='text' || L.type==='badge'){
        var lines = L.type==='badge'? [String(L.text||'').trim()] : wrap(L);
        var lh=(L.size||14)*(L.lineHeight||1.2);
        var anchor = L.align==='center'? 'middle' : (L.align==='right'? 'end' : 'start');
        var tx = L.align==='center'? L.x+L.w/2 : (L.align==='right'? L.x+L.w : L.x);
        var totalH = lines.length*lh;
        var ty = L.y + (L.type==='badge'? (L.h||0)/2 : 0);
        var tspans='';
        if(L.type==='badge'){
          tspans='<tspan x="'+tx+'" y="'+(ty+(L.size||14)*0.35)+'">'+escXml(lines[0])+'</tspan>';
          var bgR = (L.radius===999)? Math.min(L.w,(L.h||0))/2 : (L.radius||0);
          if(L.bg && L.bg!=='transparent') body.push('<rect x="'+L.x+'" y="'+L.y+'" width="'+L.w+'" height="'+(L.h||L.size*3)+'" rx="'+bgR+'" fill="'+L.bg+'" opacity="'+(L.opacity==null?1:L.opacity)+'"/>');
          if(L.border) body.push('<rect x="'+L.x+'" y="'+L.y+'" width="'+L.w+'" height="'+(L.h||L.size*3)+'" rx="'+bgR+'" fill="none" stroke="'+(L.border.split(' ').pop())+'" stroke-width="1.4"/>');
          body.push('<text x="'+tx+'" y="'+ty+'" text-anchor="middle" dominant-baseline="central" font-family="'+xmlAttr(FONTS[L.font||'sans'])+'" font-size="'+(L.size||14)+'" font-weight="'+(L.weight||800)+'" letter-spacing="'+(L.spacing||0)+'" fill="'+(L.color||'#111')+'">'+tspans+'</text>');
          return;
        }
        lines.forEach(function(ln, i){
          tspans += '<tspan x="'+tx+'" y="'+(L.y + (L.size||14)*0.82 + i*lh)+'">'+escXml(ln)+'</tspan>';
        });
        body.push('<text text-anchor="'+anchor+'" font-family="'+xmlAttr(FONTS[L.font||'sans'])+'" font-size="'+(L.size||14)+'" font-weight="'+(L.weight||700)+'" letter-spacing="'+(L.spacing||0)+'" fill="'+(L.color||'#111')+'" opacity="'+(L.opacity==null?1:L.opacity)+'">'+tspans+'</text>');
      }
    });

    return '<?xml version="1.0" encoding="UTF-8"?>\n'
      + '<svg xmlns="http://www.w3.org/2000/svg" xmlns:xlink="http://www.w3.org/1999/xlink" width="'+doc.w+'" height="'+doc.h+'" viewBox="0 0 '+doc.w+' '+doc.h+'">'
      + '<defs>'+defs.join('')+'</defs>'
      + '<rect width="100%" height="100%" fill="#ffffff"/>'
      + body.join('') + '</svg>';
  }

  function escXml(s){
    return String(s==null?'':s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
  }
  function xmlAttr(s){ return String(s).replace(/"/g,'&quot;'); }

  function fileBase(doc, extra){
    var n=(doc.name||'thiet-ke').toLowerCase();
    n = U.norm(n).replace(/[^a-z0-9]+/g,'-').replace(/^-|-$/g,'').slice(0,40) || 'thiet-ke';
    return 'canvio-'+n+(extra? '-'+extra:'')+'-'+doc.w+'x'+doc.h;
  }

  function svg(doc){
    var d = doc || Poster.doc;
    if(!d){ toast('Chưa có thiết kế'); return; }
    U.dl(fileBase(d)+'.svg', svgString(d), 'image/svg+xml');
    Board.log('Xuất SVG “'+(d.name||'')+'”');
    toast('Đã xuất SVG');
  }

  function toBlob(doc, scale, cb){
    var svgStr=svgString(doc);
    var blob=new Blob([svgStr],{type:'image/svg+xml;charset=utf-8'});
    var url=URL.createObjectURL(blob);
    var img=new Image();
    img.onload=function(){
      var canvas=document.createElement('canvas');
      canvas.width=Math.round(doc.w*(scale||2));
      canvas.height=Math.round(doc.h*(scale||2));
      var ctx=canvas.getContext('2d');
      ctx.fillStyle='#ffffff'; ctx.fillRect(0,0,canvas.width,canvas.height);
      ctx.drawImage(img,0,0,canvas.width,canvas.height);
      URL.revokeObjectURL(url);
      try{
        canvas.toBlob(function(b){ cb(b); }, 'image/png');
      }catch(e){ cb(null); }
    };
    img.onerror=function(){ URL.revokeObjectURL(url); cb(null); };
    img.src=url;
  }

  function png(doc, name){
    var d = doc || Poster.doc;
    if(!d){ toast('Chưa có thiết kế'); return; }
    toast('Đang kết xuất PNG…');
    toBlob(d, 2, function(b){
      if(!b){ toast('Không kết xuất được PNG'); return; }
      U.dl((name||fileBase(d))+'.png', b, 'image/png');
      Board.log('Xuất PNG “'+(d.name||'')+'”');
      toast('Đã xuất PNG 2×');
    });
  }

  function renderDoc(doc, format, name){
    if(format==='svg'){ U.dl(name+'.svg', svgString(doc), 'image/svg+xml'); toast('Đã xuất SVG'); return; }
    if(format==='png'){ png(doc, name); return; }
    if(format==='html'){ html(doc); return; }
    if(format==='json'){ U.dl(name+'.json', JSON.stringify(doc,null,2), 'application/json'); toast('Đã xuất JSON'); return; }
  }

  function html(doc){
    var d=doc||Poster.doc;
