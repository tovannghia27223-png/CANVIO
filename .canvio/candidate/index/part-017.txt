function readStoreZip(file){
 return file.arrayBuffer().then(function(buf){
  var bytes=new Uint8Array(buf),view=new DataView(buf),off=0,entries={};
  while(off+4<=bytes.length){
   var sig=view.getUint32(off,true);if(sig===0x02014b50||sig===0x06054b50)break;
   if(sig!==0x04034b50)throw new Error('ZIP có local header không hợp lệ tại byte '+off);
   var flags=view.getUint16(off+6,true),method=view.getUint16(off+8,true),crc=view.getUint32(off+14,true),size=view.getUint32(off+18,true),plain=view.getUint32(off+22,true),nl=view.getUint16(off+26,true),xl=view.getUint16(off+28,true);
   if(flags&0x0008)throw new Error('ZIP dùng data descriptor không được hỗ trợ trong gói portable CANVIO.');
   if(method!==0)throw new Error('ZIP dùng compression method '+method+'; chỉ hỗ trợ ZIP store do CANVIO tạo.');
   var nameStart=off+30,nameEnd=nameStart+nl,dataStart=nameEnd+xl,dataEnd=dataStart+size;
   if(dataEnd>bytes.length||plain!==size)throw new Error('ZIP entry bị cắt hoặc sai kích thước.');
   var name=new TextDecoder('utf-8').decode(bytes.subarray(nameStart,nameEnd));
   if(!name||name[0]==='/'||name.split('/').indexOf('..')>=0)throw new Error('Tên file ZIP không an toàn: '+name);
   var data=bytes.slice(dataStart,dataEnd);if(CVZip.crc32(data)!==crc)throw new Error('CRC32 không khớp: '+name);
   entries[name]=data;off=dataEnd;
  }
  if(!entries['full-backup.json'])throw new Error('ZIP không có full-backup.json.');
  return JSON.parse(new TextDecoder('utf-8').decode(entries['full-backup.json']));
 });
}
