/* ============================================================
   DESIGN MODEL + LAYOUT GENERATOR
   ============================================================ */
var Designer = (function(){

  function baseSize(w,h){ return Math.min(w,h); }

  function mkLayer(o){
    return Object.assign({
      id: U.uid('ly'), type:'rect', text:'', x:0, y:0, w:100, h:100,
      size:14, weight:700, font:'sans', color:'#101828', bg:'transparent',
      opacity:1, rotate:0, align:'left', spacing:0, radius:0, lineHeight:1.2,
      src:'', shadow:'', visible:true
    }, o);
  }

  function paletteById(id){
    for(var i=0;i<PALETTES.length;i++) if(PALETTES[i].id===id) return PALETTES[i];
    return PALETTES[0];
  }

  function shade(hex, amt, towards){ return U.mix(hex, towards||'#000000', amt); }

  /* pick accents per style */
  function styleSpec(styleId){
    var S = {
      minimal:{ headWeight:800, headTrack:-0.045, upper:true, radius:0, deco:0.34, cta:'outline', align:'left' },
      corporate:{ headWeight:800, headTrack:-0.035, upper:true, radius:6, deco:0.4, cta:'solid', align:'left' },
      editorial:{ headWeight:700, headTrack:-0.03, upper:false, radius:0, deco:0.22, cta:'outline', align:'left', font:'serif' },
      bold:{ headWeight:900, headTrack:-0.055, upper:true, radius:14, deco:0.55, cta:'solid', align:'left' },
      glass:{ headWeight:800, headTrack:-0.04, upper:true, radius:18, deco:0.48, cta:'glass', align:'left' },
      swiss:{ headWeight:800, headTrack:-0.03, upper:true, radius:0, deco:0.18, cta:'solid', align:'left', grid:true },
      luxury:{ headWeight:600, headTrack:0.02, upper:true, radius:2, deco:0.2, cta:'hairline', align:'center', font:'serif' },
      friendly:{ headWeight:800, headTrack:-0.03, upper:false, radius:22, deco:0.5, cta:'solid', align:'left', font:'rounded' },
      futuristic:{ headWeight:800, headTrack:-0.05, upper:true, radius:16, deco:0.6, cta:'glass', align:'left' }
    };
    return S[styleId] || S.corporate;
  }

  /* ---------- layout builders ---------- */
  function layoutFor(variant, ctx){
    var w=ctx.w, h=ctx.h, p=ctx.pal, sp=ctx.spec, s=ctx.scale;
    var M = Math.round(Math.min(w,h)*0.072);            /* margin */
    var layers=[];
    var dark = !!p.dark;
    var ink = p.c.ink;
    var headColor = ink;
    var subColor = p.c.muted;
    var bodyFont = sp.font || 'sans';

    /* background */
    layers.push(mkLayer({type:'rect', x:0,y:0,w:w,h:h, bg:'transparent',
      gradient:'linear-gradient(150deg,'+p.c.bg1+' 0%,'+p.c.bg2+' 100%)'}));

    /* optional image bed */
    if(ctx.image){
      layers.push(mkLayer({type:'image', x:0,y:0,w:w,h:h, src:ctx.image, opacity: dark?0.36:0.22}));
      layers.push(mkLayer({type:'rect', x:0,y:0,w:w,h:h,
        gradient:'linear-gradient(160deg,'+ (dark? 'rgba(6,12,26,.72)':'rgba(255,255,255,.82)') +' 30%, transparent 100%)'}));
    }

    var headSize = Math.round(Math.min(w,h) * (variant===0?0.118: variant===1?0.104: 0.092));
    var labelSize = Math.max(8, Math.round(Math.min(w,h)*0.0215));
    var bodySize = Math.max(9.5, Math.round(Math.min(w,h)*0.0285));
    var ctaSize = Math.max(8.5, Math.round(Math.min(w,h)*0.0235));

    if(variant===0){
      /* editorial stack: label / headline / deco / body / cta / footer */
      layers.push(mkLayer({type:'text', text:ctx.tag, x:M, y:M*0.72, w:w-M*2, h:labelSize*2,
        size:labelSize, weight:750, font:'sans', color:p.c.primary, spacing:labelSize*0.22, lineHeight:1.3}));
      layers.push(mkLayer({type:'text', text:ctx.head, x:M-M*0.04, y:Math.round(h*0.155), w:Math.round(w*0.82), h:Math.round(h*0.36),
        size:headSize, weight:sp.headWeight, font:bodyFont, color:headColor, spacing:headSize*sp.headTrack, lineHeight:0.98}));
      layers.push(mkLayer({type:'ellipse', x:Math.round(w-M-ctx.deco), y:Math.round(h*0.30), w:Math.round(ctx.deco), h:Math.round(ctx.deco),
        gradient:'radial-gradient(circle at 32% 28%, '+U.mix(p.c.accent,'#ffffff',0.55)+', '+p.c.primary+' 58%, '+shade(p.c.primary,0.35)+' 100%)',
        opacity:0.95, radius:'50%'}));
      layers.push(mkLayer({type:'line', x:M, y:Math.round(h*0.545), w:Math.round(w*0.16), h:Math.max(2,Math.round(s*3)), bg:p.c.primary}));
      layers.push(mkLayer({type:'text', text:ctx.body, x:M, y:Math.round(h*0.585), w:Math.round(w*0.6), h:Math.round(h*0.16),
        size:bodySize, weight:450, font:'sans', color:subColor, lineHeight:1.6}));
      layers.push(ctaLayer(ctx, M, Math.round(h*0.76), ctaSize, s));
      layers.push(mkLayer({type:'text', text:ctx.foot, x:M, y:Math.round(h-M*0.9), w:w-M*2, h:labelSize*2,
        size:labelSize*0.92, weight:650, font:'sans', color:subColor, spacing:labelSize*0.16, lineHeight:1.3}));
    }

    if(variant===1){
      /* split: color panel right or left */
      var pw = Math.round(w*0.42);
      var left = Math.round(Math.random()*1)===0;
      var panelX = left? 0 : w-pw;
      layers.push(mkLayer({type:'rect', x:panelX, y:0, w:pw, h:h,
        gradient:'linear-gradient(160deg,'+p.c.primary+' 0%, '+shade(p.c.primary,0.3)+' 100%)'}));
      layers.push(mkLayer({type:'ellipse', x:panelX+Math.round(pw*0.18), y:Math.round(h*0.13), w:Math.round(pw*0.64), h:Math.round(pw*0.64),
        gradient:'radial-gradient(circle at 34% 30%, rgba(255,255,255,.72), rgba(255,255,255,.06) 62%)', opacity:0.9, radius:'50%'}));
      layers.push(mkLayer({type:'ellipse', x:panelX-Math.round(pw*0.16), y:Math.round(h*0.62), w:Math.round(pw*0.5), h:Math.round(pw*0.5),
        gradient:'radial-gradient(circle at 50% 40%, '+p.c.accent+', '+shade(p.c.accent,0.3)+')', opacity:0.92, radius:'50%'}));
      var tx = left? pw+M*0.9 : M;
      var tw = w-M*2-pw*0.55;
      layers.push(mkLayer({type:'text', text:ctx.tag, x:tx, y:M*0.72, w:tw, h:labelSize*2,
        size:labelSize, weight:750, color:p.c.primary, spacing:labelSize*0.22, lineHeight:1.3}));
      layers.push(mkLayer({type:'text', text:ctx.head, x:tx-M*0.04, y:Math.round(h*0.16), w:tw, h:Math.round(h*0.42),
        size:headSize, weight:sp.headWeight, font:bodyFont, color:headColor, spacing:headSize*sp.headTrack, lineHeight:0.98}));
      layers.push(mkLayer({type:'text', text:ctx.body, x:tx, y:Math.round(h*0.63), w:tw, h:Math.round(h*0.17),
        size:bodySize, weight:450, color:subColor, lineHeight:1.6}));
      layers.push(ctaLayer(ctx, tx, Math.round(h*0.81), ctaSize, s));
    }

    if(variant===2){
      /* centered poster */
      layers.push(mkLayer({type:'ellipse', x:Math.round(w/2-ctx.deco*0.85), y:Math.round(h*0.10), w:Math.round(ctx.deco*1.7), h:Math.round(ctx.deco*1.7),
        gradient:'radial-gradient(circle at 40% 32%, '+U.mix(p.c.accent,'#ffffff',0.5)+', '+p.c.primary+' 62%, '+shade(p.c.primary,0.4)+')', opacity:0.2, radius:'50%'}));
      layers.push(mkLayer({type:'text', text:ctx.tag, x:M, y:Math.round(h*0.09), w:w-M*2, h:labelSize*2,
        size:labelSize, weight:750, color:p.c.primary, spacing:labelSize*0.28, align:'center', lineHeight:1.3}));
      layers.push(mkLayer({type:'text', text:ctx.head, x:Math.round(w*0.06), y:Math.round(h*0.27), w:Math.round(w*0.88), h:Math.round(h*0.3),
        size:Math.round(headSize*0.94), weight:sp.headWeight, font:bodyFont, color:headColor, spacing:headSize*sp.headTrack, align:'center', lineHeight:1.02}));
      layers.push(mkLayer({type:'text', text:ctx.body, x:Math.round(w*0.16), y:Math.round(h*0.60), w:Math.round(w*0.68), h:Math.round(h*0.13),
        size:bodySize, weight:450, color:subColor, align:'center', lineHeight:1.65}));
      var cw = Math.min(w-M*2, Math.round(w*0.5));
      var ctaC = ctaLayer(ctx, Math.round(w/2-cw/2), Math.round(h*0.755), ctaSize, s);
      ctaC.w = cw; ctaC.align='center';
      layers.push(ctaC);
      layers.push(mkLayer({type:'text', text:ctx.foot, x:M, y:Math.round(h-M*0.9), w:w-M*2, h:labelSize*2,
        size:labelSize*0.92, weight:650, color:subColor, spacing:labelSize*0.18, align:'center', lineHeight:1.3}));
    }

    if(variant===3){
      /* band: solid headline band */
      var bandH = Math.round(h*0.46);
      layers.push(mkLayer({type:'rect', x:0, y:0, w:w, h:bandH,
        gradient:'linear-gradient(135deg,'+p.c.primary+', '+shade(p.c.primary,0.28)+')'}));
      layers.push(mkLayer({type:'text', text:ctx.tag, x:M, y:Math.round(bandH*0.14), w:w-M*2, h:labelSize*2,
        size:labelSize, weight:800, color:'rgba(255,255,255,.86)', spacing:labelSize*0.24, lineHeight:1.3}));
      layers.push(mkLayer({type:'text', text:ctx.head, x:M-M*0.04, y:Math.round(bandH*0.32), w:Math.round(w*0.8), h:Math.round(bandH*0.55),
        size:headSize, weight:sp.headWeight, font:bodyFont, color:'#ffffff', spacing:headSize*sp.headTrack, lineHeight:0.98}));
      layers.push(mkLayer({type:'ellipse', x:Math.round(w-M-ctx.deco*0.8), y:Math.round(bandH-ctx.deco*0.42), w:Math.round(ctx.deco*0.9), h:Math.round(ctx.deco*0.9),
        gradient:'radial-gradient(circle at 36% 30%, '+U.mix(p.c.accent,'#ffffff',0.4)+', '+p.c.accent+')', opacity:0.95, radius:'50%'}));
      layers.push(mkLayer({type:'text', text:ctx.body, x:M, y:Math.round(bandH+M*0.85), w:Math.round(w*0.66), h:Math.round(h*0.16),
        size:bodySize, weight:450, color:subColor, lineHeight:1.6}));
      layers.push(ctaLayer(ctx, M, Math.round(h*0.78), ctaSize, s));
      layers.push(mkLayer({type:'line', x:M, y:Math.round(h*0.72), w:Math.round(w-M*2), h:1, bg:U.mix(p.c.muted,'#ffffff',0.7)}));
      layers.push(mkLayer({type:'text', text:ctx.foot, x:M, y:Math.round(h-M*0.85), w:w-M*2, h:labelSize*2,
        size:labelSize*0.9, weight:650, color:subColor, spacing:labelSize*0.18, lineHeight:1.3}));
    }

    if(variant===4){
      /* diagonal / poster block */
      layers.push(mkLayer({type:'rect', x:Math.round(w*0.52), y:Math.round(h*0.06), w:Math.round(w*0.54), h:Math.round(h*0.34),
        gradient:'linear-gradient(135deg,'+p.c.primary+','+p.c.accent+')', opacity:0.92, radius: Math.round(w*0.04)}));
      layers.push(mkLayer({type:'text', text:ctx.tag, x:M, y:M*0.8, w:Math.round(w*0.5), h:labelSize*2,
        size:labelSize, weight:800, color:p.c.primary, spacing:labelSize*0.24, lineHeight:1.3}));
      layers.push(mkLayer({type:'text', text:ctx.head, x:M-M*0.04, y:Math.round(h*0.28), w:Math.round(w*0.86), h:Math.round(h*0.35),
        size:Math.round(headSize*1.02), weight:sp.headWeight, font:bodyFont, color:headColor, spacing:headSize*sp.headTrack, lineHeight:0.96}));
      layers.push(mkLayer({type:'line', x:M, y:Math.round(h*0.68), w:Math.round(w*0.12), h:Math.max(2,Math.round(s*3)), bg:p.c.accent}));
      layers.push(mkLayer({type:'text', text:ctx.body, x:M, y:Math.round(h*0.715), w:Math.round(w*0.62), h:Math.round(h*0.14),
        size:bodySize, weight:450, color:subColor, lineHeight:1.6}));
      layers.push(ctaLayer(ctx, M, Math.round(h*0.855), ctaSize, s));
      layers.push(mkLayer({type:'text', text:ctx.foot, x:Math.round(w*0.5), y:Math.round(h-M*0.85), w:Math.round(w*0.44), h:labelSize*2,
        size:labelSize*0.88, weight:650, color:subColor, spacing:labelSize*0.16, align:'right', lineHeight:1.3}));
    }

    return layers;
  }

  function ctaLayer(ctx, x, y, size, s){
    var w = Math.round(Math.max(ctx.w*0.34, ctx.cta.length*size*0.72));
    var h = Math.round(size*3.1);
    var sp = ctx.spec;
    var base = {type:'badge', text:'  '+ctx.cta+'  ', x:x, y:y, w:w, h:h, size:size, weight:800,
      font:'sans', spacing:size*0.14, radius: sp.radius>10? 999 : (sp.radius||0), lineHeight:1};
    if(sp.cta==='solid'){ base.bg = ctx.pal.c.primary; base.color = ctx.pal.dark? '#08111f' : '#ffffff'; }
    else if(sp.cta==='glass'){ base.bg = ctx.pal.dark? 'rgba(255,255,255,.16)' : U.mix(ctx.pal.c.primary,'#ffffff',0.86); base.color = ctx.pal.dark? '#ffffff': ctx.pal.c.primary; }
    else if(sp.cta==='outline'){ base.bg='transparent'; base.color=ctx.pal.c.primary; base.border='1.5px solid '+ctx.pal.c.primary; }
    else if(sp.cta==='hairline'){ base.bg='transparent'; base.color=ctx.pal.c.ink; base.border='1px solid '+ctx.pal.c.ink; base.radius=0; }
    return base;
  }

  /* ---------- prompt parsing ---------- */
  function parse(prompt){
    var n = ' '+U.norm(prompt).replace(/[^a-z0-9]+/g,' ').trim()+' ';
    var out = { industry:'marketing', style:'corporate', palette:'auto', mood:'', kind:'poster', ratio:'portrait', raw:prompt||'' };

    /* industry: keyword length counts more (more specific), earlier mention counts more */
    var bestKey='marketing', bestScore=0;
    Object.keys(INDUSTRY).forEach(function(k){
      var score=0;
      var words=[k].concat(INDUSTRY[k].words||[]);
      words.forEach(function(w){
        var nw=' '+U.norm(w)+' ';
        var at=n.indexOf(nw);
        if(nw.trim() && at>-1) score += Math.max(2, nw.trim().length) * (at<22? 1.6 : 1);
      });
      if(score>bestScore){ bestScore=score; bestKey=k; }
    });
    out.industry = bestKey;
    out.industryScore = Math.round(bestScore*10)/10;
    var styleMap = {
      minimal:['toi gian','minimal','sach se','don gian'], corporate:['doanh nghiep','corporate','chuyen nghiep','cong ty'],
      editorial:['bien tap','editorial','tap chi','magazine'], bold:['dam manh','bold','noi bat','manh'],
      glass:['kinh','glass','trong suot','mo'], swiss:['swiss','luoi','grid','thuy si'],
      luxury:['cao cap','luxury','sang trong','premium','sang'],
      friendly:['than thien','friendly','tre trung','vui','mem'],
      futuristic:['tuong lai','futuristic','hien dai','tech','neo','vu tru']
    };
    Object.keys(styleMap).forEach(function(sid){
      styleMap[sid].forEach(function(kw){ if(n.indexOf(kw)>-1) out.style=sid; });
    });
    PALETTES.forEach(function(p){
      if(n.indexOf(U.norm(p.id))>-1) out.palette=p.id;
      p.tags.forEach(function(t){ if(n.indexOf(U.norm(t))>-1 && out.palette==='auto') out.palette=p.id; });
    });
    if(/website|landing|trang web|\bweb\b/.test(n)) out.kind='web';
    if(/\bbo\b|\bset\b|da kich thuoc|nhieu size|nhieu kich thuoc/.test(n)) out.kind='set';
    if(/nhan dien|thuong hieu|\bbrand\b|identity/.test(n)) out.kind='identity';
    if(/\bstory\b|9:16|doc dung/.test(n)) out.ratio='story';
    else if(/vuong|1:1|square/.test(n)) out.ratio='square';
    else if(/ngang|landscape|3:2/.test(n)) out.ratio='landscape';
    else if(/a4\b/.test(n)) out.ratio='a4';
    out.label = out.label || null;
    var pal2 = (out.palette && out.palette!=='auto')? paletteById(out.palette) : autoPalette(out.industry);
    out.paletteName = pal2.name;
    out.industryLabel = (INDUSTRY[out.industry]||{}).label || out.industry;
    return out;
  }

  function content(ctx){
    var ind = INDUSTRY[ctx.industry] || INDUSTRY.marketing;
    var seed = (ctx.raw||'').length;
    var heads = ind.heads, idx = seed % heads.length;
    var head = ctx.headline || heads[idx];
    var body = ctx.bodyText || ind.body;
    var tag = ctx.tagText || ind.tag;
    var cta = ctx.ctaText || ind.cta;
    var brand = ctx.brandName || 'NORTHSTAR';
    return {
      head: head, body: body, tag: tag, cta: cta,
      foot: brand + ' · ' + ind.label.toUpperCase() + ' · 2027'
    };
  }

  function autoPalette(industry){
    var ind = INDUSTRY[industry] || INDUSTRY.marketing;
    var words = [ind.label].concat(ind.words||[]).map(U.norm);
    var best=null, bestScore=-1;
    PALETTES.forEach(function(p){
      var score=0;
      p.tags.forEach(function(t){
        var nt=U.norm(t);
        words.forEach(function(w){
          if(!w) return;
          if(nt.indexOf(w)>-1 || w.indexOf(nt)>-1) score+=2;
        });
      });
      if(score>bestScore){ bestScore=score; best=p; }
    });
    return best || PALETTES[0];
  }

  function build(opts){
    var w = opts.w, h = opts.h;
    var pal = (opts.palette && opts.palette!=='auto') ? paletteById(opts.palette) : autoPalette(opts.industry);
    if(!pal) pal = PALETTES[0];
    var spec = styleSpec(opts.style);
    var scale = Math.min(w,h)/440;
    var base = {w:w,h:h,pal:pal,spec:spec,scale:scale,image:opts.image||null};
    var txt = content(opts);
    var ctx = Object.assign({}, base, txt);
    ctx.deco = Math.round(Math.min(w,h) * (0.24 + spec.deco*0.36));
    var variant = opts.variant==null? 0 : opts.variant;
    var layers = layoutFor(variant, ctx);
    return {
      id: U.uid('doc'), name: (opts.name||'Thiết kế'), w:w, h:h, ratio: opts.ratio||'portrait',
      palette: pal.id, style: opts.style||'corporate', layers: layers, industry: opts.industry||'marketing',
      variant: variant, kind: opts.kind||'poster', sizeName: opts.sizeName||null,
      createdAt: new Date().toISOString(),
      meta: { head:txt.head, body:txt.body, tag:txt.tag, cta:txt.cta, brand:ctx.brandName||'NORTHSTAR' }
    };
  }

  return {
    build: build,
    parse: parse,
    autoPalette: autoPalette,
    paletteById: paletteById,
    styleSpec: styleSpec,
    mkLayer: mkLayer,
    content: content,
    INDUSTRY: INDUSTRY
  };
})();

  function bindDrag(el, L){

      if(L.locked) return;

      var rect = stageEl().getBoundingClientRect();
      var z = rect.width / doc.w;
      var resizing = e.offsetX > el.offsetWidth-16 && e.offsetY > el.offsetHeight-16;
      var ow=L.w, oh=L.h;
      function mv(ev){
        if(resizing){
          L.w = Math.max(12, Math.round(ow + (ev.clientX-startX)/z));
          L.h = Math.max(6, Math.round(oh + (ev.clientY-startY)/z));
          el.style.width=L.w+'px'; el.style.height=L.h+'px';
        } else {
          var nx = Math.round(ox + (ev.clientX-startX)/z);
          var ny = Math.round(oy + (ev.clientY-startY)/z);
          if(snap){
            var g = Math.round(Math.min(doc.w,doc.h)*0.072);
            nx = snapVal(nx, [g, Math.round(doc.w/2 - L.w/2), doc.w - L.w - g], Math.round(doc.w*0.01));
            ny = snapVal(ny, [g, Math.round(doc.h/2 - L.h/2), doc.h - L.h - g], Math.round(doc.h*0.01));
          }
          L.x = U.clamp(nx, -L.w*0.5, doc.w-L.w*0.5);
          L.y = U.clamp(ny, -L.h*0.5, doc.h-L.h*0.5);
          el.style.left=L.x+'px'; el.style.top=L.y+'px';
          syncPos();
        }
      }
      function up(){
        document.removeEventListener('mousemove', mv);
        document.removeEventListener('mouseup', up);
        History.push(); Project.autosave();
      }
      document.addEventListener('mousemove', mv);
      document.addEventListener('mouseup', up);
    });
  }

    if(w) w.style.backgroundSize = showGrid? '22px 22px':'';
  }

  function setZoom(z){
    zoom=U.clamp(z,0.2,2);
    var st=stageEl();
    if(st) st.style.transform='scale('+zoom+')';
    var wrap=document.getElementById('stageScale');
    if(wrap && doc){ wrap.style.left='0px'; wrap.style.top='0px'; wrap.style.width=(doc.w*zoom)+'px'; wrap.style.height=(doc.h*zoom)+'px'; }
    var zl=document.getElementById('zoomLabel'); if(zl) zl.textContent=Math.round(zoom*100)+'%';
  }

  function zoomStep(d){ setZoom(zoom+d); }

  function fit(){
    var wrap=document.getElementById('canvasWrap');
    var st=stageEl();
    if(!wrap||!st||!doc) return;
    var availW=wrap.clientWidth-56, availH=Math.max(360, window.innerHeight-320);
    var z=Math.min(availW/doc.w, availH/doc.h, 1);
    setZoom(z>0? z : 0.5);
  }

  function upload(ev){
    var f=ev.target.files && ev.target.files[0];
    if(!f||!doc) return;
    var r=new FileReader();
    r.onload=function(){
      var L=Designer.mkLayer({type:'image', src:r.result, x:Math.round(doc.w*0.1), y:Math.round(doc.h*0.1), w:Math.round(doc.w*0.4), h:Math.round(doc.w*0.4)});
      doc.layers.push(L); sel=L.id; render(); History.push(); Project.autosave();
      toast('Đã thêm ảnh');
    };
    r.readAsDataURL(f);
    ev.target.value='';
  }

  function applyPalette(id){
    if(!doc) return;
    var pal=Designer.paletteById(id);
    var old=Designer.paletteById(doc.palette);
    doc.palette=id;
    doc.layers.forEach(function(L, i){
      if(i===0 && L.gradient){ L.gradient='linear-gradient(150deg,'+pal.c.bg1+' 0%,'+pal.c.bg2+' 100%)'; }
      if(L.type==='text'||L.type==='badge'){
        if(L.color && L.color.toLowerCase()===old.c.ink.toLowerCase()) L.color=pal.c.ink;
        if(L.bg && L.bg.toLowerCase()===old.c.primary.toLowerCase()) L.bg=pal.c.primary;
      }
      if(L.type==='line') L.bg=pal.c.primary;
      if(L.type==='rect' && L.bg && L.bg.toLowerCase()===old.c.primary.toLowerCase()) L.bg=pal.c.primary;
      if(L.type==='ellipse' && L.bg && L.bg.toLowerCase()===old.c.accent.toLowerCase()) L.bg=pal.c.accent;
    });
    render(); History.push(); Project.autosave();
    toast('Đã áp bảng màu '+pal.name);
  }

  /* ---------- history ---------- */
  function snapshot(){ return JSON.stringify(doc); }

  var History = {
    push: function(){
      if(!doc) return;
      var s=snapshot();
      if(undoS.length && undoS[undoS.length-1]===s) return;
      undoS.push(s); redoS=[];
      if(undoS.length>60) undoS.shift();
    },
    undo: function(){
      if(undoS.length<2) { toast('Không còn thao tác để hoàn tác'); return; }
      var cur=undoS.pop(); redoS.push(cur);
      doc=JSON.parse(undoS[undoS.length-1]);
      sel=null; render(); fit(); Project.autosave();
      toast('Đã hoàn tác');
    },
    redo: function(){
      if(!redoS.length){ toast('Không có thao tác để làm lại'); return; }
      var s=redoS.pop(); undoS.push(s);
      doc=JSON.parse(s); sel=null; render(); fit(); Project.autosave();
      toast('Đã làm lại');
    }
  };

  function qa(){
    var rows=[];
    var L=doc? doc.layers : [];
    var texts=L.filter(function(x){return x.type==='text';});
    var ctas=L.filter(function(x){return x.type==='badge';});
    rows.push(L.length? {ok:true,msg:'Có '+L.length+' lớp trên khung'} : {ok:false,msg:'Khung thiết kế đang trống'});
    rows.push(texts.length? {ok:true,msg:'Có '+texts.length+' lớp chữ'} : {ok:false,msg:'Chưa có tiêu đề — thêm lớp chữ'});
    rows.push(ctas.length? {ok:true,msg:'Có nút kêu gọi hành động'} : {ok:false,msg:'Chưa có nút kêu gọi — thêm nhãn'});
    var over = L.some(function(x){ return x.x<0 || x.y<0 || x.x+x.w>doc.w+2 || x.y+x.h>doc.h+2; });
    rows.push(over? {ok:false,msg:'Có lớp tràn ra ngoài khung'} : {ok:true,msg:'Không có lớp tràn khung'});
    var contrast = checkContrast();
    rows.push(contrast>=4.5? {ok:true,msg:'Tương phản chữ/nền đạt '+contrast.toFixed(1)+':1'} : {ok:false,msg:'Tương phản thấp ('+contrast.toFixed(1)+':1) — nên tăng'});
    var list=document.getElementById('qaList');
    if(list){
      list.innerHTML=rows.map(function(r){
        return '<div class="advice '+(r.ok?'ok':'fix')+'"><span class="ico">'+(r.ok?'✓':'✕')+'</span><span>'+r.msg+'</span></div>';
      }).join('');
    }
    openModal('modalQA');
  }

  function checkContrast(){
    if(!doc) return 1;
    var pal=Designer.paletteById(doc.palette);
    var textL=doc.layers.filter(function(x){return x.type==='text';});
    if(!textL.length) return U.contrast(pal.c.ink, pal.c.bg1);
    var head=textL.slice().sort(function(a,b){ return (b.size||0)-(a.size||0); })[0];
    var c=head.color||pal.c.ink;
    if(!/^#|^rgb/i.test(c)) c=pal.c.ink;
    return U.contrast(c, pal.c.bg1);
  }

  function newDesign(){
    var r = doc? doc.ratio : 'portrait';
    setDoc(blankDoc(r));
    toast('Đã tạo thiết kế mới');
  }

  function init(){
    if(!doc) setDoc(blankDoc('portrait'));
    bindProps();
    var st=stageEl();
    if(st) st.addEventListener('mousedown', function(e){ if(e.target===st) select(null); });
    window.addEventListener('resize', U.debounce(function(){ if(document.getElementById('page-poster').classList.contains('active')) fit(); }, 160));
  }

  return {
    init: init, get doc(){ return doc; }, setDoc: setDoc, render: render,
    add: add, duplicate: duplicate, remove: remove, front: front, back: back,
    setRatio: setRatio, toggleSnap: toggleSnap, toggleGrid: toggleGrid,
    zoom: zoomStep, zoomFit: fit, fit: fit, upload: upload, applyPalette: applyPalette,
    qa: qa, newDesign: newDesign, history: History, checkContrast: checkContrast,
    select: select, get selected(){ return current(); }, snapshot: function(){ return doc; }, history: History,
    getLayers: function(){ return doc ? doc.layers.map(function(L,i){ if(!L.id)L.id=U.uid('ly'); return {id:L.id,name:L.name||L.text||({image:'Ảnh',rect:'Hình chữ nhật',ellipse:'Hình tròn',line:'Đường',badge:'Nhãn'}[L.type]||'Lớp')+'',type:L.type,visible:L.visible!==false,locked:!!L.locked,index:i,selected:L.id===sel}; }) : []; },
    selectLayer: function(id){ var L=doc&&doc.layers.find(function(x){return x.id===id}); if(!L||L.locked)return false; select(id); return true; },
    renameLayer: function(id,name){ var L=doc&&doc.layers.find(function(x){return x.id===id}); if(!L)return false; L.name=String(name||'').slice(0,100); render(); History.push(); Project.autosave(); return true; },
    setLayerVisible: function(id,on){ var L=doc&&doc.layers.find(function(x){return x.id===id}); if(!L)return false; L.visible=!!on; render(); History.push(); Project.autosave(); return true; },
    setLayerLocked: function(id,on){ var L=doc&&doc.layers.find(function(x){return x.id===id}); if(!L)return false; L.locked=!!on; if(on&&sel===id)sel=null; render(); History.push(); Project.autosave(); return true; },
    moveLayer: function(id,where){ if(!doc)return false; var i=doc.layers.findIndex(function(x){return x.id===id}); if(i<0)return false; var L=doc.layers.splice(i,1)[0]; if(where==='front')doc.layers.push(L); else doc.layers.unshift(L); render(); History.push(); Project.autosave(); return true; },
    duplicateLayer: function(id){ if(!doc)return false; var L=doc&&doc.layers.find(function(x){return x.id===id}); if(!L||L.locked)return false; select(id); duplicate(); return true; },
    deleteLayer: function(id){ var L=doc&&doc.layers.find(function(x){return x.id===id}); if(!L||L.locked)return false; select(id); remove(); return true; },
    snapshotState: function(){ return doc ? JSON.stringify(doc) : null; },
    restoreState: function(raw){ try{var d=typeof raw==='string'?JSON.parse(raw):raw; if(!d||!Array.isArray(d.layers))return false; setDoc(d); return true;}catch(e){return false;} }
  };
})();

/* ============================================================
   STYLER — single source of truth for layer CSS (editor + exports)
   ============================================================ */
var Styler = (function(){
  function css(L){
    var s=[];
    s.push('position:absolute');
    s.push('left:'+L.x+'px');
    s.push('top:'+L.y+'px');
    s.push('width:'+L.w+'px');
    if(L.h) s.push('height:'+L.h+'px');
    s.push('opacity:'+(L.opacity==null?1:L.opacity));
    s.push('transform:rotate('+(L.rotate||0)+'deg)');
    if(L.gradient) s.push('background:'+L.gradient);
    else if(L.type==='ellipse'||L.type==='rect'||L.type==='badge'||L.type==='line') s.push('background:'+(L.bg||'transparent'));
    if(L.border) s.push('border:'+L.border);
    if(L.type==='ellipse') s.push('border-radius:50%');
    else if(L.radius) s.push('border-radius:'+(L.radius===999?'999px':L.radius+'px'));
    if(L.type==='text'||L.type==='badge'){
      s.push('font-family:'+FONTS[L.font||'sans']);
      s.push('font-size:'+(L.size||14)+'px');
      s.push('font-weight:'+(L.weight||700));
      s.push('line-height:'+(L.lineHeight||1.2));
      s.push('letter-spacing:'+(L.spacing||0)+'px');
      s.push('color:'+(L.color||'#111'));
      s.push('text-align:'+(L.align||'left'));
      if(L.type==='badge'){ s.push('display:grid'); s.push('place-items:center'); s.push('white-space:nowrap'); }
      else { s.push('white-space:pre-wrap'); s.push('word-break:break-word'); }
    }
    return s.join(';');
  }
  function html(L){
    if(L.type==='image' && L.src){
      return '<div style="'+css(L)+'"><img src="'+L.src+'" alt="" style="width:100%;height:100%;object-fit:cover;display:block"></div>';
    }
    if(L.type==='text'||L.type==='badge') return '<div style="'+css(L)+'">'+U.esc(L.text||'')+'</div>';
    return '<div style="'+css(L)+'"></div>';
  }
  function all(doc){ return doc.layers.map(html).join(''); }
  return {css:css, html:html, all:all};
})();

/* ============================================================
   PREVIEW — scaled thumbnails for cards / grids
   ============================================================ */
var Preview = (function(){
  function thumb(doc, boxW, boxH, radius){
    if(!doc) return '';
    var z = Math.min(boxW/doc.w, boxH/doc.h);
    var w = Math.round(doc.w*z), h = Math.round(doc.h*z);
    return '<div style="width:'+boxW+'px;height:'+boxH+'px;display:grid;place-items:center;overflow:hidden;position:relative;background:var(--surface-3)">'
      + '<div style="width:'+w+'px;height:'+h+'px;overflow:hidden;position:relative;background:#fff;box-shadow:0 2px 8px rgba(16,32,60,.12)">'
      + '<div style="position:absolute;left:0;top:0;width:'+doc.w+'px;height:'+doc.h+'px;transform:scale('+z+');transform-origin:top left">'
      + Styler.all(doc) + '</div></div></div>';
  }
  function card(doc, boxW, boxH, label, sub, actions){
    return '<div class="pvCard">'
      + '<div class="pvThumb" style="height:'+boxH+'px">'+thumb(doc, boxW, boxH)+'</div>'
      + '<div class="pvBody"><div><b>'+U.esc(label||'')+'</b><small>'+U.esc(sub||'')+'</small></div>'
      + '<div class="row" style="gap:4px">'+(actions||'')+'</div></div></div>';
  }
  return {thumb:thumb, card:card};
})();
